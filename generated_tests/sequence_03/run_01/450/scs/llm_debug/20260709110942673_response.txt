package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class DateParseTest {

    private String baseUrl;

    @Before
    public void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testValidDayMondayAndValidMonthJanuary() {
        given()
            .pathParam("dayname", "mon")
            .pathParam("monthname", "jan")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testValidDayTuesdayAndValidMonthFebruary() {
        given()
            .pathParam("dayname", "tue")
            .pathParam("monthname", "feb")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testValidDayWednesdayAndValidMonthMarch() {
        given()
            .pathParam("dayname", "wed")
            .pathParam("monthname", "mar")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testValidDayThursdayAndValidMonthApril() {
        given()
            .pathParam("dayname", "thur")
            .pathParam("monthname", "apr")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testValidDayFridayAndValidMonthMay() {
        given()
            .pathParam("dayname", "fri")
            .pathParam("monthname", "may")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testInvalidDayAndInvalidMonth() {
        given()
            .pathParam("dayname", "invalid")
            .pathParam("monthname", "invalid")
        .when()
            .get("/api/dateparse/{dayname}/{monthname}")
        .then()
            .statusCode(500);
    }
}
