package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static io.restassured.RestAssured.baseURI;
import static org.hamcrest.Matchers.lessThan;

public class FisherTest {

    static {
        baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testFisherReturnsZeroWhenPLessThanZero() {
        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/10/5/-0.1")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherReturnsOneWhenPGreaterThanOne() {
        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/2/2/1e15")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherReturnsPWhenPBetweenZeroAndOne() {
        given()
            .when()
            .get("/api/fisher/1/1/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherWithOddMOddN() {
        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/1/1/0.75")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherWithOddMEvenN() {
        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/1/2/0.75")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherWithEvenMEvenN() {
        given()
            .when()
            .get("/api/fisher/10/5/0.75")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/api/fisher/2/4/0.75")
            .then()
            .statusCode(200);
    }
}
