package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.not;

public class ConstraintRequiresTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCreateRequiresConstraint() {
        String productName = "Prod-Req-1-" + UUID.randomUUID().toString();
        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));

        Response response = given()
                .formParam("sourceFeature", "SourceFeat")
                .formParam("requiredFeature", "RequiredFeat")
                .when()
                .post("/products/{productName}/constraints/requires", productName);

        response.then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testEvaluateConstraintSourceActiveRequiredInactive() {
        String productName = "Prod-Req-2-" + UUID.randomUUID().toString();
        String configName = "Config1";
        String sourceFeat = "SourceFeat";
        String requiredFeat = "RequiredFeat";

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeat).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeat).then().statusCode(lessThan(300));
        given().formParam("sourceFeature", sourceFeat).formParam("requiredFeature", requiredFeat).when().post("/products/{productName}/constraints/requires", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, sourceFeat).then().statusCode(lessThan(300));

        Response response = given().when().get("/products/{productName}/configurations/{configurationName}/features", productName, configName);

        response.then().body("$", hasItem(requiredFeat));
    }

    @Test(timeout = 60000)
    public void testEvaluateConstraintSourceInactive() {
        String productName = "Prod-Req-3-" + UUID.randomUUID().toString();
        String configName = "Config1";
        String sourceFeat = "SourceFeat";
        String requiredFeat = "RequiredFeat";

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeat).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeat).then().statusCode(lessThan(300));
        given().formParam("sourceFeature", sourceFeat).formParam("requiredFeature", requiredFeat).when().post("/products/{productName}/constraints/requires", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));

        Response response = given().when().get("/products/{productName}/configurations/{configurationName}/features", productName, configName);

        response.then().body("$", not(hasItem(requiredFeat)));
    }

    @Test(timeout = 60000)
    public void testEvaluateConstraintSourceAndRequiredActive() {
        String productName = "Prod-Req-4-" + UUID.randomUUID().toString();
        String configName = "Config1";
        String sourceFeat = "SourceFeat";
        String requiredFeat = "RequiredFeat";

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeat).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeat).then().statusCode(lessThan(300));
        given().formParam("sourceFeature", sourceFeat).formParam("requiredFeature", requiredFeat).when().post("/products/{productName}/constraints/requires", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", productName, configName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, sourceFeat).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", productName, configName, requiredFeat).then().statusCode(lessThan(300));

        Response response = given().when().get("/products/{productName}/configurations/{configurationName}/features", productName, configName);

        response.then().body("$", hasItem(requiredFeat));
    }
}
