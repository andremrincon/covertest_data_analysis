package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class BessjTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url", System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080");
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testNLessThan2ReturnsBadRequest() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 1, 2.5);
        r.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testAxZeroReturnsOk() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 3, 0);
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAxGreaterThanNUsesAsymptoticBranch() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 3, 5.0);
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAxMuchSmallerThanNUsesBackwardRecurrence() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 10, 1e-10);
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNegativeXAndOddNProducesNegativeValue() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 3, -2.5);
        r.then().body("value", lessThan(0.0f));
    }

    @Test(timeout = 60000)
    public void testNonNumericNReturnsBadRequest() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", "abc", 2.5);
        r.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testBessj0ElseBranchForLargeX() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 3, 10.0);
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessj1ElseBranchWithNegativeX() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 4, -10.0);
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testLargeNAndLargeXProducesOk() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response r = given().when().get("/api/bessj/{n}/{x}", 50, 1.0e10);
        r.then().statusCode(200);
    }
}
