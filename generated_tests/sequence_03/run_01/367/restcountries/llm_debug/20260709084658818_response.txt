package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ContributionTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testContributeAcceptedWithValidPayload() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        given().contentType(ContentType.JSON).body("{\"amount\":1000,\"currency\":\"EUR\",\"token\":\"" + token + "\"}")
                .when().post("/contribute")
                .then().statusCode(202);
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestMissingAmount() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        given().contentType(ContentType.JSON).body("{\"currency\":\"EUR\",\"token\":\"" + token + "\"}")
                .when().post("/contribute")
                .then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestNegativeAmount() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        given().contentType(ContentType.JSON).body("{\"amount\":-10,\"currency\":\"EUR\",\"token\":\"" + token + "\"}")
                .when().post("/contribute")
                .then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestInvalidTokenFormat() {
        given().when().get("/v1/alpha?codes=US,CA").then().statusCode(lessThan(300));
        given().contentType(ContentType.JSON).body("{\"amount\":50,\"currency\":\"EUR\",\"token\":\"123\"}")
                .when().post("/contribute")
                .then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestMissingCurrency() {
        given().when().get("/v1/alpha/GB").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        given().contentType(ContentType.JSON).body("{\"amount\":200,\"token\":\"" + token + "\"}")
                .when().post("/contribute")
                .then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestEmptyBody() {
        given().when().get("/v1/alpha?codes=US").then().statusCode(lessThan(300));
        given().contentType(ContentType.JSON).body("")
                .when().post("/contribute")
                .then().statusCode(400);
    }
}
