public class CountryRestV1Test {

        baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;

    public void testGetByAlpha_NotFound() {
        given()
            .contentType(ContentType.JSON)

            .statusCode(404);
