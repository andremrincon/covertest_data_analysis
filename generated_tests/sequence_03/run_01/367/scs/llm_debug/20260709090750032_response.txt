package ts01gpt_5_mini;

import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class TitleTest {

    private static String BASE;

    @BeforeClass
    public static void init() {
        String prop = System.getProperty("base.url");
        String env = System.getenv("BASE_URL");
        if (prop != null && !prop.isEmpty()) {
            BASE = prop;
        } else if (env != null && !env.isEmpty()) {
            BASE = env;
        } else {
            BASE = "http://localhost:8080";
        }
    }

    @Test(timeout = 60000)
    public void testMaleValidTitleReturns1() {
        given().when().get(BASE + "/api/pat/arrange-test").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/male/mr");
        assertEquals("1", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testMaleInvalidTitleReturnsMinus1() {
        given().when().get(BASE + "/api/pat/arrange-" + System.currentTimeMillis()).then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/male/queen");
        assertEquals("-1", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testFemaleValidTitleReturns0() {
        given().when().get(BASE + "/api/pat/setup-female").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/female/mrs");
        assertEquals("0", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testNoneValidTitleReturns2() {
        given().when().get(BASE + "/api/pat/setup-none").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/none/dr");
        assertEquals("2", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testUnknownSexReturnsMinus1() {
        given().when().get(BASE + "/api/pat/setup-unknown").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/unknown/dr");
        assertEquals("-1", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testNeuterTriggersServerErrorReturns500() {
        given().when().get(BASE + "/api/pat/setup-neuter").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/title/neuter/Jones");
        act.then().statusCode(500);
    }
}
