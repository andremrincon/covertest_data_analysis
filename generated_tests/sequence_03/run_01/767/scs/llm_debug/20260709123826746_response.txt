package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class NotyPevarTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchXPlusYEquals56() {
        given().when().get("/api/notypevar/0/a").then().statusCode(lessThan(300));
        Response response = given().when().get("/api/notypevar/28/a");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchXsPlusYEqualsHello7() {
        given().when().get("/api/notypevar/0/a").then().statusCode(lessThan(300));
        Response response = given().when().get("/api/notypevar/7/a");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchXsCompareToSLessThan0() {
        given().when().get("/api/notypevar/0/a").then().statusCode(lessThan(300));
        Response response = given().when().get("/api/notypevar/2/world");
        response.then().statusCode(200);
    }
}
