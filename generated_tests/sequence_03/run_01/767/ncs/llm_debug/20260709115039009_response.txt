package ts01glm_5_2;

import static io.restassured.RestAssured.given;

public class NcsRestTest {

    public void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testClassifyInvalidEdgeA() {
        given()
            .baseUri(baseUrl)

            .statusCode(200);
