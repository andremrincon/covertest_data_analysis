package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CalcTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCalcPlus() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));

        given()
            .pathParam("op", "plus")
            .pathParam("arg1", 15.5)
            .pathParam("arg2", 4.5)
        .when()
            .get("/api/calc/{op}/{arg1}/{arg2}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCalcInvalidArg() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));

        given()
            .pathParam("op", "plus")
            .pathParam("arg1", 10)
            .pathParam("arg2", "twenty")
        .when()
            .get("/api/calc/{op}/{arg1}/{arg2}")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testCalcDivideByZero() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));

        given()
            .pathParam("op", "divide")
            .pathParam("arg1", 100)
            .pathParam("arg2", 0)
        .when()
            .get("/api/calc/{op}/{arg1}/{arg2}")
        .then()
            .statusCode(500);
    }
}
