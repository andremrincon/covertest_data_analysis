package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class TriangleClassificationTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testIsoscelesTriangle() {
        given()
            .when()
                .get("/api/triangle/3/3/4")
            .then()
                .statusCode(200)
                .body("type", equalTo("isosceles"));
    }

    @Test(timeout = 60000)
    public void testEquilateralTriangle() {
        given()
            .when()
                .get("/api/triangle/3/3/3")
            .then()
                .statusCode(200)
                .body("type", equalTo("equilateral"));
    }

    @Test(timeout = 60000)
    public void testScaleneTriangle() {
        given()
            .when()
                .get("/api/triangle/3/4/5")
            .then()
                .statusCode(200)
                .body("type", equalTo("scalene"));
    }

    @Test(timeout = 60000)
    public void testInvalidTriangleZeroSide() {
        given()
            .when()
                .get("/api/triangle/0/4/5")
            .then()
                .statusCode(200)
                .body("type", equalTo("invalid"));
    }

    @Test(timeout = 60000)
    public void testInvalidTriangleNotATriangle() {
        given()
            .when()
                .get("/api/triangle/1/2/10")
            .then()
                .statusCode(200)
                .body("type", equalTo("invalid"));
    }
}
