package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class GammqTest {

    private String baseUrl;

    @Before
    public void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGcfExecutionPath() {
        Response response = given()
                .pathParam("a", 0.001)
                .pathParam("x", 1000.0)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserExecutionPath() {
        Response response = given()
                .pathParam("a", 5.5)
                .pathParam("x", 2.3)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserWithZeroX() {
        Response response = given()
                .pathParam("a", 5.5)
                .pathParam("x", 0.0)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testInvalidNegativeX() {
        Response response = given()
                .pathParam("a", 5.5)
                .pathParam("x", -1.0)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testInvalidNegativeA() {
        Response response = given()
                .pathParam("a", -1.0)
                .pathParam("x", 2.3)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testInvalidParameterType() {
        Response response = given()
                .pathParam("a", "abc")
                .pathParam("x", 2.3)
                .when()
                .get("/api/gammq/{a}/{x}");

        response.then().statusCode(400);
    }
}
