package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.hasItems;
import static org.hamcrest.Matchers.equalTo;

public class FeatureTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) base = System.getenv("API_BASE");
        if (base == null || base.isEmpty()) base = "http://localhost:8080/";
        RestAssured.baseURI = base.endsWith("/") ? base : base + "/";
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProductReturns201() {
        String uid = UUID.randomUUID().toString();
        String product = "Smartwatch-Series-" + uid;
        String feature = "Blood-Oxygen-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "Measures SpO2 on demand.")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForProductContainsAddedFeature() {
        String uid = UUID.randomUUID().toString();
        String product = "AeroBook-Pro-" + uid;
        String feature = "Backlit-Keyboard-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "Comfortable typing experience.")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().when().get("/products/{productName}/features", product)
                .then().body("featureName", hasItems(feature));
    }

    @Test(timeout = 60000)
    public void testUpdateFeatureReturns200() {
        String uid = UUID.randomUUID().toString();
        String product = "EcoTablet-" + uid;
        String feature = "stylus-support-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "Initial desc")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "RGB backlit keyboard with customizable zones and per-key lighting.")
                .when().put("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDeleteFeatureReturns204() {
        String uid = UUID.randomUUID().toString();
        String product = "SmartWatch-Pro-" + uid;
        String feature = "heart-rate-monitor-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .formParam("description", "Heart rate monitoring.")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().when().delete("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(204);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationReturns201() {
        String uid = UUID.randomUUID().toString();
        String product = "QuantumLeap-AI-Platform-" + uid;
        String config = "standard-gpu-cluster-" + uid;
        String feature = "distributed-training-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, config)
                .then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature)
                .then().statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetConfigurationFeaturesReturns200AndContainsFeature() {
        String uid = UUID.randomUUID().toString();
        String product = "ultra-laptop-" + uid;
        String config = "us-standard-16gb-" + uid;
        String feature = "wifi-6e-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, config)
                .then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature)
                .then().statusCode(lessThan(300));
        given().when().get("/products/{productName}/configurations/{configurationName}/features", product, config)
                .then().body("$", hasItems(feature));
    }

    @Test(timeout = 60000)
    public void testDeleteFeatureFromConfigurationReturns204() {
        String uid = UUID.randomUUID().toString();
        String product = "QuantumLeap-AI-" + uid;
        String config = "Premium-Tier-" + uid;
        String feature = "Realtime-Analytics-Dashboard-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, config)
                .then().statusCode(lessThan(300));
        given().contentType("application/x-www-form-urlencoded")
                .when().post("/products/{productName}/features/{featureName}", product, feature)
                .then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature)
                .then().statusCode(lessThan(300));
        given().when().delete("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature)
                .then().statusCode(204);
    }

    @Test(timeout = 60000)
    public void testGetProductByNameReturns200() {
        String uid = UUID.randomUUID().toString();
        String product = "Smartwatch-Series-" + uid;
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().get("/products/{productName}", product).then().statusCode(200);
    }
}
