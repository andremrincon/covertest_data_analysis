package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class WrongProductConfigurationExceptionTest {

    @BeforeClass
    public static void setup() {
        String base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) {
            base = System.getProperty("base.url", "http://localhost:8080");
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForMalformedProductNameReturns500() {
        String uniqueProduct = "p-" + UUID.randomUUID().toString();
        given().pathParam("productName", uniqueProduct).when().post("/products/{productName}").then().statusCode(lessThan(300));
        Response act = given().pathParam("productName", "product/with/slashes").when().get("/products/{productName}/features");
        act.then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProductWithProblematicNamesReturns500() {
        String uniqueProduct = "p-" + UUID.randomUUID().toString();
        given().pathParam("productName", uniqueProduct).when().post("/products/{productName}").then().statusCode(lessThan(300));
        Response act = given().pathParam("productName", "Gaming-PC-Tower").pathParam("featureName", "UltraBook-Z1").when().post("/products/{productName}/features/{featureName}");
        act.then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationWithJsonLikeFeatureNameReturns500() {
        String product = "p-" + UUID.randomUUID().toString();
        String config = "c-" + UUID.randomUUID().toString();
        given().pathParam("productName", product).when().post("/products/{productName}").then().statusCode(lessThan(300));
        given().pathParam("productName", product).pathParam("configurationName", config).when().post("/products/{productName}/configurations/{configurationName}").then().statusCode(lessThan(300));
        String problematicFeature = "{\"feature\":\"new-feature\"}";
        Response act = given().pathParam("productName", product).pathParam("configurationName", config).pathParam("featureName", problematicFeature).when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}");
        act.then().statusCode(500);
    }
}
