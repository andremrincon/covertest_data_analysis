package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class SampleControllerTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSayHelloWithProvidedName() {
        String unique = "arrange-" + UUID.randomUUID().toString();
        given().queryParam("name", unique).when().get("/").then().statusCode(lessThan(300));
        given().queryParam("name", "María-José O'Connor-Smith III").when().get("/").then().body(equalTo("Hello María-José O'Connor-Smith III!!"));
    }

    @Test(timeout = 60000)
    public void testSayHelloWithDefaultName() {
        String unique = "arrange-" + UUID.randomUUID().toString();
        given().queryParam("name", unique).when().get("/").then().statusCode(lessThan(300));
        given().when().get("/").then().body(equalTo("Hello Guest!!"));
    }

    @Test(timeout = 60000)
    public void testSlowApiWithPositiveDelay() {
        String unique = "arrange-" + UUID.randomUUID().toString();
        given().queryParam("name", unique).when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", 1).when().get("/slowApi").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiWithZeroDelayUsesRandom() {
        String unique = "arrange-" + UUID.randomUUID().toString();
        given().queryParam("name", unique).when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", 0).when().get("/slowApi").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSlowApiWithNegativeDelayProducesServerError() {
        String unique = "arrange-" + UUID.randomUUID().toString();
        given().queryParam("name", unique).when().get("/").then().statusCode(lessThan(300));
        given().queryParam("delay", -1).when().get("/slowApi").then().statusCode(500);
    }
}
