package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrencyFields() {
        given()
            .when()
                .get("/v2/alpha/US")
            .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2AlphaWithFieldsCurrenciesReturnsCurrencyData() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/alpha/DE")
            .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCountriesWithCurrency() {
        given()
            .when()
                .get("/v2/currency/EUR")
            .then()
                .statusCode(200)
                .body("size()", greaterThan(0));
    }

    @Test(timeout = 60000)
    public void testV2CurrencyWithFieldsCurrenciesReturnsCurrencyCode() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/currency/USD")
            .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2RegionalblocReturnsCountriesWithCurrency() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/regionalbloc/EU")
            .then()
                .statusCode(200)
                .body("size()", greaterThan(0));
    }

    @Test(timeout = 60000)
    public void testV2AllWithCurrenciesFieldReturnsCurrencyData() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/all")
            .then()
                .statusCode(200)
                .body("size()", greaterThan(0));
    }

    @Test(timeout = 60000)
    public void testV2AlphaMultipleCodesReturnsCurrencyData() {
        given()
            .queryParam("codes", "US,CA,MX")
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/alpha")
            .then()
                .statusCode(200)
                .body("size()", greaterThan(0));
    }

    @Test(timeout = 60000)
    public void testV2NameReturnsCurrencyFields() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/name/France")
            .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CapitalReturnsCurrencyFields() {
        given()
            .queryParam("fields", "name;currencies")
            .when()
                .get("/v2/capital/London")
            .then()
                .statusCode(200)
                .body("currencies", notNullValue());
    }
}
