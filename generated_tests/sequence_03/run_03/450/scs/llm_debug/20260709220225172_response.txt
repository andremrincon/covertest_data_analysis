package ts01gpt_5_mini;

import org.junit.Test;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {
    private static final String BASE;
    static {
        String env = System.getenv("API_BASE");
        if (env != null && !env.isEmpty()) {
            BASE = env;
        } else {
            BASE = System.getProperty("api.base", "http://localhost:8080");
        }
    }

    @Test(timeout = 60000)
    public void test_costfuns_minus4_abab_returns_0() {
        given().when().get(BASE + "/api/text2txt/The/quick/brown").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/-4/abab");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_costfuns_i5_zzzz_results_in_10() {
        given().when().get(BASE + "/api/pat/The").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/5/zzzz");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_costfuns_i_minus500_abab_triggers_negative_branches() {
        given().when().get(BASE + "/api/calc/add/1/2").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/-500/abab");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_costfuns_i667_ababba_exercises_compareto_equal_zero_case() {
        given().when().get(BASE + "/api/pat/ABABCABAB/ABAB").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/667/ababba");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_costfuns_i555_abab_preserves_upper_numeric_branch_before_string_checks() {
        given().when().get(BASE + "/api/pat/The").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/555/abab");
        act.then().body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void test_costfuns_minus4_baab_executes_string_equals_branch() {
        given().when().get(BASE + "/api/title/male/Smith").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/costfuns/-4/baab");
        act.then().body(equalTo("10"));
    }
}
