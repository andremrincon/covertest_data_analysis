package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class RemainderTest {

    private final String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");

    @Test(timeout = 60000)
    public void testRemainderPositiveAPositiveB() {

        int a = 17;
        int b = 5;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)
                .pathParam("b", b)
                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveANegativeB() {

        int a = 17;
        int b = -5;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)
                .pathParam("b", b)
                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeAPositiveB() {

        int b = 5;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)
                .pathParam("b", b)
                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeANegativeB() {

        int a = -17;
        int b = -5;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)
                .pathParam("b", b)
                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderZeroA() {

        int a = 0;
        int b = 5;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)
                .pathParam("b", b)
                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderZeroB() {

        int a = 5;
        int b = 0;


        Response response = given()
                .baseUri(baseUrl)
                .pathParam("a", a)

                .when()
                .get("/api/remainder/{a}/{b}");


        response.then().statusCode(400);
    }
}
