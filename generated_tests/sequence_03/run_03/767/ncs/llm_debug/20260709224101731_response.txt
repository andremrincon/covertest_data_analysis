package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class BessjTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void bessj_nLessThan2_throwsException_n0() {
        given()
                .pathParam("n", 0)
                .pathParam("x", 2.5)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void bessj_nLessThan2_throwsException_n1() {
        given()
                .pathParam("n", 1)
                .pathParam("x", 2.5)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void bessj_axGreaterThanN_returnsResult() {
        given()
                .pathParam("n", 3)
                .pathParam("x", 2.5)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_axEqualsZero_returnsZero() {
        given()
                .pathParam("n", 3)
                .pathParam("x", 0.0)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_negativeX_oddN_returnsNegativeAns() {
        given()
                .pathParam("n", 3)
                .pathParam("x", -2.5)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_axLessThanOrEqualN_elseBranch() {
        given()
                .pathParam("n", 4)
                .pathParam("x", 1.0)
                .accept("application/json")
                .when()
                .get("/api/bessj/{n}/{x}")
                .then()
                .statusCode(200);
    }
}
