package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class Text2TxtTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        RestAssured.baseURI = baseUrl != null ? baseUrl : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testTwo() {
        Response response = given().when().get("/api/text2txt/two/any/any");
        response.then().body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testFor() {
        Response response = given().when().get("/api/text2txt/for/any/any");
        response.then().body(equalTo("4"));
    }

    @Test(timeout = 60000)
    public void testYou() {
        Response response = given().when().get("/api/text2txt/you/any/any");
        response.then().body(equalTo("u"));
    }

    @Test(timeout = 60000)
    public void testAnd() {
        Response response = given().when().get("/api/text2txt/and/any/any");
        response.then().body(equalTo("n"));
    }

    @Test(timeout = 60000)
    public void testSeeYou() {
        Response response = given().when().get("/api/text2txt/see/you/any");
        response.then().body(equalTo("cu"));
    }

    @Test(timeout = 60000)
    public void testByTheWay() {
        Response response = given().when().get("/api/text2txt/by/the/way");
        response.then().body(equalTo("btw"));
    }
}
