package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.nullValue;

public class CORSFilterTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl != null && !baseUrl.isEmpty()) {
            RestAssured.baseURI = baseUrl;
        } else {
            RestAssured.baseURI = "http://localhost:8080/rest";
        }
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsAccessControlAllowOriginHeaderOnV1All() {
        given()
                .when()
                .get("/v1/all")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsAccessControlAllowMethodsHeaderOnV1Alpha() {
        given()
                .queryParam("codes", "US")
                .when()
                .get("/v1/alpha")
                .then()
                .statusCode(lessThan(300))
                .header("Access-Control-Allow-Methods", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsAccessControlAllowHeadersHeaderOnV1AlphaCode() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Headers", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsCacheControlHeaderOnV1Currency() {
        given()
                .when()
                .get("/v1/currency/USD")
                .then()
                .statusCode(404)
                .header("Cache-Control", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsAllCorsHeadersOnV1Name() {
        given()
                .when()
                .get("/v1/name/France")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue())
                .header("Access-Control-Allow-Methods", nullValue())
                .header("Access-Control-Allow-Headers", nullValue())
                .header("Cache-Control", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV1Callingcode() {
        given()
                .when()
                .get("/v1/callingcode/1")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV1Capital() {
        given()
                .when()
                .get("/v1/capital/London")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV1Region() {
        given()
                .when()
                .get("/v1/region/Europe")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV2All() {
        given()
                .when()
                .get("/v2/all")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV2AlphaCode() {
        given()
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV2Name() {
        given()
                .when()
                .get("/v2/name/Germany")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void corsFilter_doFilter_addsHeadersOnV2Region() {
        given()
                .when()
                .get("/v2/region/Europe")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", nullValue());
    }
}
