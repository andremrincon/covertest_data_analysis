package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

public class NcsRestTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testFisher_mExceedsLimit_returns400() {
        given()
            .accept("application/json")
        .when()
            .get("/api/fisher/1001/5/0.75")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testFisher_nExceedsLimit_returns400() {
        given()
            .accept("application/json")
        .when()
            .get("/api/fisher/10/1001/0.75")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testFisher_validParams_returns200() {
        given()
            .accept("application/json")
        .when()
            .get("/api/fisher/10/5/0.75")
        .then()
            .statusCode(200)
            .body("resultAsDouble", notNullValue());
    }

    @Test(timeout = 60000)
    public void testRemainder_aExceedsUpperLimit_returns400() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/10001/5")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainder_bExceedsLowerLimit_returns400() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/17/-10001")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainder_validParams_returns200() {
        given()
            .accept("application/json")
        .when()
            .get("/api/remainder/17/5")
        .then()
            .statusCode(200)
            .body("resultAsInt", notNullValue());
    }
}
