package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class NotyPevarTest {

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchI0TrueAndI3True() {
        given()
                .pathParam("i", 28)
                .pathParam("s", "a")
                .when()
                .get("/api/notypevar/{i}/{s}")
                .then()
                .statusCode(200)
                .body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchI1TrueAndI2True() {
        given()
                .pathParam("i", 7)
                .pathParam("s", "world")
                .when()
                .get("/api/notypevar/{i}/{s}")
                .then()
                .statusCode(200)
                .body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevarBranchI3False() {
        given()
                .pathParam("i", 3)
                .pathParam("s", "a")
                .when()
                .get("/api/notypevar/{i}/{s}")
                .then()
                .statusCode(200)
                .body(equalTo("0"));
    }
}
