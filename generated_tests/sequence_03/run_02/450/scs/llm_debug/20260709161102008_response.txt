package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testCostfuns_iEquals5() {
        given()
            .when()
            .get("/api/costfuns/5/test")
            .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iLessThanMinus444() {
        given()
            .when()
            .get("/api/costfuns/-500/test")
            .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iGreaterThan666() {
        given()
            .when()
            .get("/api/costfuns/700/test")
            .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iEqualsMinus4AndSEqualsBaab() {
        given()
            .when()
            .get("/api/costfuns/-4/baab")
            .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iZeroAndSLessThanAbabba() {
        given()
            .when()
            .get("/api/costfuns/0/a")
            .then()
            .statusCode(200)
            .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iZeroAndSEqualsAbab() {
        given()
            .when()
            .get("/api/costfuns/0/abab")
            .then()
            .statusCode(200)
            .body(equalTo("6"));
    }
}
