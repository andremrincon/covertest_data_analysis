package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class RemainderTest {

    @BeforeClass
    public static void init() {
        String env = System.getenv("BASE_URL");
        String base = env != null ? env : System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAIsZeroReturnsBadRequest() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/0/5").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testBIsZeroReturnsBadRequest() {
        given().when().get("/api/triangle/1/1/1").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/5/0").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testPositiveAAndPositiveBProducesPositiveRemainder() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/17/5").then().body("result", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testPositiveAAndNegativeBProducesComputedRemainder() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/17/-9").then().body("result", equalTo(8));
    }

    @Test(timeout = 60000)
    public void testNegativeAAndPositiveBProducesNegativeRemainder() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/-17/5").then().body("result", equalTo(-2));
    }

    @Test(timeout = 60000)
    public void testNegativeAAndNegativeBProducesPositiveRemainder() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/remainder/-17/-9").then().body("result", equalTo(8));
    }
}
