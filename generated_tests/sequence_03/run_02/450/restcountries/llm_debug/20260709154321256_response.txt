package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.Optional;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class LanguageTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = Optional.ofNullable(System.getenv("BASE_URL")).orElse("http://localhost:8080/rest");
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSetIso6391ViaV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].iso639_1", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void testSetIso6392ViaV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].iso639_2", equalTo("eng"));
    }

    @Test(timeout = 60000)
    public void testSetNameViaV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].name", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testSetNativeNameViaV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().body("languages[0].nativeName", equalTo("English"));
    }

    @Test(timeout = 60000)
    public void testSetIso6391ViaV1Name_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].languages[0].iso639_1", equalTo("fr"));
    }

    @Test(timeout = 60000)
    public void testSetIso6392ViaV1Name_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].languages[0].iso639_2", equalTo("fra"));
    }

    @Test(timeout = 60000)
    public void testSetNameViaV1Name_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].languages[0].name", equalTo("French"));
    }

    @Test(timeout = 60000)
    public void testSetNativeNameViaV1Name_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France");
        resp.then().body("[0].languages[0].nativeName", equalTo("français"));
    }
}
