package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;

public class BessjTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testBessjWithNLessThan2Returns400() {
        given()
            .get(BASE_URL + "/api/bessj/1/2.5")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testBessjWithXZeroReturns200() {
        given()
            .get(BASE_URL + "/api/bessj/3/0.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjForwardRecurrenceWithAxLessThan8() {
        given()
            .get(BASE_URL + "/api/bessj/2/5.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjForwardRecurrenceWithAxGreaterOrEqual8() {
        given()
            .get(BASE_URL + "/api/bessj/2/10.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjForwardRecurrenceWithNegativeXAndAxGreaterOrEqual8() {
        given()
            .get(BASE_URL + "/api/bessj/2/-10.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjForwardRecurrenceWithNegativeXAndOddN() {
        given()
            .get(BASE_URL + "/api/bessj/3/-10.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjBackwardRecurrence() {
        given()
            .get(BASE_URL + "/api/bessj/10/2.5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBessjWithInvalidNReturns400() {
        given()
            .get(BASE_URL + "/api/bessj/abc/2.5")
        .then()
            .statusCode(400);
    }
}
