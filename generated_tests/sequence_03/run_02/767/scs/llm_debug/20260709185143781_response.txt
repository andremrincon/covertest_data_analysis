package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.junit.Assert.*;

public class CookieTest {

    private static final String BASE_URL = System.getProperty("baseUrl", "http://localhost:8080");

    @BeforeClass
    public static void setUp() {
        RestAssured.baseURI = BASE_URL;
    }

    @Test(timeout = 60000)
    public void testUserIdCookieWithUserPrefixReturnsOne() {
        given()
            .when()
                .get("/api/cookie/userid/user12345/example.com")
            .then()
                .statusCode(lessThan(300));

        Response response = given()
            .when()
                .get("/api/cookie/userid/user12345/example.com");

        assertEquals(200, response.statusCode());
        assertTrue(response.body().asString().contains("1"));
    }

    @Test(timeout = 60000)
    public void testSessionCookieWithAmAndAbcComReturnsOne() {
        given()
            .when()
                .get("/api/cookie/session/am/abc.com")
            .then()
                .statusCode(lessThan(300));

        Response response = given()
            .when()
                .get("/api/cookie/session/am/abc.com");

        assertEquals(200, response.statusCode());
        assertTrue(response.body().asString().contains("1"));
    }

    @Test(timeout = 60000)
    public void testSessionCookieWithDifferentValuesReturnsTwo() {
        given()
            .when()
                .get("/api/cookie/session/xyz/other.com")
            .then()
                .statusCode(lessThan(300));

        Response response = given()
            .when()
                .get("/api/cookie/session/xyz/other.com");

        assertEquals(200, response.statusCode());
        assertTrue(response.body().asString().contains("2"));
    }
}
