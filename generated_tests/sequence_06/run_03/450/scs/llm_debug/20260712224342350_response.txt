import io.restassured.RestAssured;
import org.junit.Test;

import java.util.Arrays;
import java.util.List;

public class DateParseTest {

    private final String baseUrl = "http://localhost:8080";

    @Test(timeout = 60000)
    public void testDateParseDaysGroup1() {
        List<String> days = Arrays.asList("tue", "thur", "fri");
        for (String day : days) {
            RestAssured.given()
                    .when()
                    .get(baseUrl + "/api/dateparse/" + day + "/jan")
                    .then()
                    .statusCode(200);
        }
    }
