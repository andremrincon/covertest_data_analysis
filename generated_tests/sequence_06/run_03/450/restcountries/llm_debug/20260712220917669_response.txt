package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setUp() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080/rest";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAlphaCodes_semicolon_unique() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().param("codes", "US;CA;US").when().get("/v1/alpha");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAlpha_single_alpha2_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAlpha_single_invalid_numeric_returns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/123");
        act.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testName_fullText_exact_France() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().param("fullText", true).when().get("/v1/name/France");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testName_substring_United_false() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().param("fullText", false).when().get("/v1/name/United");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCallingCode_valid_1() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/callingcode/1");
        act.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCapital_London_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/capital/London");
        act.then().statusCode(200);
    }
}
