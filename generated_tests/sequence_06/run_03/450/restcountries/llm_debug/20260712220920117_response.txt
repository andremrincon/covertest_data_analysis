package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CountryTranslationsTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSetDe_viaV1Alpha_US_1() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        CountryTranslations t = act.jsonPath().getObject("translations", CountryTranslations.class);
        assertEquals("Vereinigte Staaten von Amerika", t.getDe());
    }

    @Test(timeout = 60000)
    public void testSetEs_viaV1Alpha_US_2() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        CountryTranslations t = act.jsonPath().getObject("translations", CountryTranslations.class);
        assertEquals("Estados Unidos", t.getEs());
    }

    @Test(timeout = 60000)
    public void testSetFr_viaV1Alpha_US_3() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        CountryTranslations t = act.jsonPath().getObject("translations", CountryTranslations.class);
        assertEquals("\u00c9tats-Unis", t.getFr());
    }

    @Test(timeout = 60000)
    public void testSetJa_viaV1Alpha_US_4() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        CountryTranslations t = act.jsonPath().getObject("translations", CountryTranslations.class);
        assertEquals("\u30a2\u30e1\u30ea\u30ab\u5408\u8846\u56fd", t.getJa());
    }

    @Test(timeout = 60000)
    public void testSetIt_viaV1Alpha_US_5() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response act = given().when().get("/v1/alpha/US");
        CountryTranslations t = act.jsonPath().getObject("translations", CountryTranslations.class);
        assertEquals("Stati Uniti D'America", t.getIt());
    }
}

class CountryTranslations {
    private String de;
    private String es;
    private String fr;
    private String ja;
    private String it;

    public CountryTranslations() {
    }

    public String getDe() {
        return de;
    }

    public void setDe(String de) {
        this.de = de;
    }

    public String getEs() {
        return es;
    }

    public void setEs(String es) {
        this.es = es;
    }

    public String getFr() {
        return fr;
    }

    public void setFr(String fr) {
        this.fr = fr;
    }

    public String getJa() {
        return ja;
    }

    public void setJa(String ja) {
        this.ja = ja;
    }

    public String getIt() {
        return it;
    }

    public void setIt(String it) {
        this.it = it;
    }
}
