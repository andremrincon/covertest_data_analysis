import org.junit.Test;
import static io.restassured.RestAssured.given;

        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(404);
    }

    public void testV1CapitalNotFound() {
        given()
            .pathParam("capital", "123")
        .when()

        .then()
            .statusCode(404);
    }

    public void testV2CurrencyBadRequest() {

}
