package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class NotFoundExceptionMapperTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("baseUrl");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void toResponse_nonExistentPath_returnsNotFound() {
        given()
                .when()
                .get("/v1/nonexistentpath")
                .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void toResponse_nonExistentSubPath_returnsNotFound() {
        given()
                .when()
                .get("/v2/nonexistentpath")
                .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void toResponse_rootNonExistentPath_returnsNotFound() {
        given()
                .when()
                .get("/totallynonexistentendpoint")
                .then()
                .statusCode(404);
    }
}
