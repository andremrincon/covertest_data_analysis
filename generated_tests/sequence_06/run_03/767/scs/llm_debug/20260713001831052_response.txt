package ts01glm_5_2;

import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class Ordered4Test {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testIncreasingOrder() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/aaaaa/bbbbb/ddddd/ccccc")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDecreasingOrder() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/ddddd/ccccc/aaaaa/bbbbb")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedWithValidLengths() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/aaaaa/bbbbb/bbbbb/aaaaa")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedWithShortLength() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/a/bbbbb/ccccc/ddddd")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnorderedWithLongLength() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/aaaaaaa/bbbbb/ccccc/ddddd")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testIncreasingOrderWithLengthSixBoundary() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/api/ordered4/aaaaaa/bbbbbb/dddddd/cccccc")
        .then()
            .statusCode(200);
    }
}
