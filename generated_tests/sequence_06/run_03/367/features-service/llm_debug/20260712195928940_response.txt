package ts01glm_5_2;

import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

import org.junit.Before;
import org.junit.Test;

public class EvaluationResultTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testEvaluationResultInitializationViaConfigurationFeaturesEndpoint() {
        String productName = "EvalTestProduct-" + java.util.UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "base-config";
        String featureName = "test-feature";

        given().when()
            .post("/products/" + productName)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/features/" + featureName)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/configurations/" + configurationName)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/configurations/" + configurationName + "/features/" + featureName)
        .then().statusCode(lessThan(300));

        given().when()
            .get("/products/" + productName + "/configurations/" + configurationName + "/features")
        .then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluationResultInitializationViaGetConfigurationEndpoint() {
        String productName = "EvalTestProduct2-" + java.util.UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "standard-config";

        given().when()
            .post("/products/" + productName)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/configurations/" + configurationName)
        .then().statusCode(lessThan(300));

        given().when()
            .get("/products/" + productName + "/configurations/" + configurationName)
        .then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluationResultInitializationWithRequiresConstraint() {
        String productName = "EvalTestProduct3-" + java.util.UUID.randomUUID().toString().substring(0, 8);
        String configurationName = "constrained-config";
        String sourceFeature = "source-feat";
        String requiredFeature = "required-feat";

        given().when()
            .post("/products/" + productName)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/features/" + sourceFeature)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/features/" + requiredFeature)
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/configurations/" + configurationName)
        .then().statusCode(lessThan(300));

        given()
            .formParam("sourceFeature", sourceFeature)
            .formParam("requiredFeature", requiredFeature)
        .when()
            .post("/products/" + productName + "/constraints/requires")
        .then().statusCode(lessThan(300));

        given().when()
            .post("/products/" + productName + "/configurations/" + configurationName + "/features/" + sourceFeature)
        .then().statusCode(lessThan(300));

        given().when()
            .get("/products/" + productName + "/configurations/" + configurationName + "/features")
        .then().statusCode(200);
    }
}
