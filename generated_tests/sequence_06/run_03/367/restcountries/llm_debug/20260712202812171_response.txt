package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import io.restassured.http.ContentType;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ResponseEntityTest {
    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAllowOriginHeaderOnV1Alpha() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response r = given().when().get("/v1/alpha/US");
        r.then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testAllowMethodsHeaderOnV1All() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        Response r = given().when().get("/v1/all");
        r.then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testAllowHeadersHeaderOnV1AlphaQuery() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response r = given().queryParam("codes", "US,CA").when().get("/v1/alpha");
        r.then().header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnV1Currency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response r = given().when().get("/v1/currency/USD");
        r.then().header("Cache-Control", "public, max-age=86400");
    }

    @Test(timeout = 60000)
    public void testV1NameEndpointReturns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response r = given().when().get("/v1/name/France");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testContributePostAccepted() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String payload = "{\"amount\":10,\"currency\":\"USD\",\"token\":\"" + UUID.randomUUID().toString() + "\"}";
        Response r = given().contentType(ContentType.JSON).body(payload).when().post("/contribute");
        r.then().statusCode(202);
    }

    @Test(timeout = 60000)
    public void testV2AlphaFieldsStatus200() {
        given().queryParam("codes", "US,CA").when().get("/v2/alpha").then().statusCode(lessThan(300));
        Response r = given().queryParam("fields", "name;capital;population").when().get("/v2/alpha/US");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRootGetHasCorsHeader() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response r = given().when().get("/");
        r.then().header("Access-Control-Allow-Origin", "*");
    }
}
