package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CORSFilterTest {

    @Test(timeout = 60000)
    public void testCorsAllowOriginHeader() {
        given()
            .baseUri("http://localhost:8080/rest")
        .when()
            .get("/v1/all")
        .then()
            .statusCode(200)
            .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowMethodsHeader() {
        given()
            .baseUri("http://localhost:8080/rest")
        .when()
            .get("/v1/alpha/US")
        .then()
            .statusCode(200)
            .header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowHeadersHeader() {
        given()
            .baseUri("http://localhost:8080/rest")
        .when()
            .get("/v1/name/France")
        .then()
            .statusCode(200)
            .header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"));
    }

    @Test(timeout = 60000)
    public void testCacheControlHeader() {
        given()
            .baseUri("http://localhost:8080/rest")
        .when()
            .get("/v1/capital/London")
        .then()
            .statusCode(200)
            .header("Cache-Control", equalTo("public, max-age=86400"));
    }
}
