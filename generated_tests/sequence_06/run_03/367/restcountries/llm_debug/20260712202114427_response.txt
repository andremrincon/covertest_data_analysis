package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class ContributionTest {

    private static final String BASE_URL = System.getProperty("baseUrl", "http://localhost:8080/rest");

    @BeforeClass
    public static void setUp() {
        RestAssured.baseURI = BASE_URL;
    }

    @Test(timeout = 60000)
    public void getByAlpha_validAlphaCode_returns200() {
        given()
                .accept(ContentType.JSON)
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByAlpha_invalidAlphaCodeTooLong_returns400() {
        given()
                .accept(ContentType.JSON)
                .when()
                .get("/v1/alpha/1234")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByAlpha_notFoundAlphaCode_returns404() {
        given()
                .accept(ContentType.JSON)
                .when()
                .get("/v1/alpha/XYZ")
                .then()
                .statusCode(404);
    }

    public void getByAlphaList_validCodes_returns200() {
