package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class ContributionTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void getByAlpha_returnsNull_whenAlpha2CodeNotFound() {
        given()
            .when()
                .get("/v1/alpha/XX")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByAlpha_returnsNull_whenAlpha3CodeNotFound() {
        given()
            .when()
                .get("/v1/alpha/XXX")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByAlpha_returnsCountry_whenValidAlpha2Code() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByAlpha_returnsCountry_whenValidAlpha3Code() {
        given()
            .when()
                .get("/v1/alpha/USA")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByCodeList_handlesNullCodesParam() {
        given()
            .when()
                .get("/v1/alpha")
            .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByCodeList_deduplicates_whenDuplicateCodesProvided() {
        given()
            .queryParam("codes", "US;US")
            .when()
                .get("/v1/alpha")
            .then()
                .body("size()", is(1));
    }

    @Test(timeout = 60000)
    public void getByCodeList_returnsMultipleCountries_whenDistinctCodesProvided() {
        given()
            .queryParam("codes", "US;CA")
            .when()
                .get("/v1/alpha")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByCodeList_returnsNotFound_whenAllCodesInvalid() {
        given()
            .queryParam("codes", "XX;YY;ZZ")
            .when()
                .get("/v1/alpha")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByCodeList_returnsPartialResults_whenMixOfValidAndInvalidCodes() {
        given()
            .queryParam("codes", "US;XX")
            .when()
                .get("/v1/alpha")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void fulltextSearch_matchesByAltSpelling() {
        given()
            .queryParam("fullText", "true")
            .when()
                .get("/v1/name/French Republic")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void fulltextSearch_matchesByExactName() {
        given()
            .queryParam("fullText", "true")
            .when()
                .get("/v1/name/France")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void fulltextSearch_returnsNotFound_whenNameNotMatched() {
        given()
            .queryParam("fullText", "true")
            .when()
                .get("/v1/name/NonExistentCountryXYZ")
            .then()
                .statusCode(404);
    }

    @Test(timeout = 60000)
    public void substringSearch_matchesByAltSpellingSubstring() {
        given()
            .when()
                .get("/v1/name/Republic")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void substringSearch_matchesByNameSubstring() {
        given()
            .when()
                .get("/v1/name/United")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void loadJson_dataSuccessfullyLoaded() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .statusCode(200);
    }
}
