package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class ExpintTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testExpintWithXGreaterThanOne() {
        Response response = given()
                .when()
                .get("/api/expint/3/2.5");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintWithXLessThanOrEqualToOne() {
        Response response = given()
                .when()
                .get("/api/expint/1/0.1");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpintWithNegativeN() {
        Response response = given()
                .when()
                .get("/api/expint/-1/2.5");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpintWithNegativeX() {
        Response response = given()
                .when()
                .get("/api/expint/3/-999.9");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpintWithNZeroAndXZero() {
        Response response = given()
                .when()
                .get("/api/expint/0/0.0");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpintWithNOneAndXZero() {
        Response response = given()
                .when()
                .get("/api/expint/1/0.0");
        response.then().statusCode(400);
    }
}
