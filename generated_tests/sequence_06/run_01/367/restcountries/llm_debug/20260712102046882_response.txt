package ts01glm_5_2;

import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import org.junit.Before;
import org.junit.Test;

public class CurrencyTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV1AlphaReturnsCurrencyCode() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200)
                .body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaReturnsCurrencyName() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200)
                .body("currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaReturnsCurrencySymbol() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .statusCode(200)
                .body("currencies[0].symbol", equalTo("$"));
    }

    @Test(timeout = 60000)
    public void testV1CurrencyReturnsCurrencyCode() {
        given()
            .when()
                .get("/v1/currency/USD")
            .then()
                .statusCode(200)
                .body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testV1CurrencyReturnsCurrencyName() {
        given()
            .when()
                .get("/v1/currency/EUR")
            .then()
                .statusCode(200)
                .body("currencies[0].name", equalTo("Euro"));
    }

    @Test(timeout = 60000)
    public void testV1CurrencyReturnsCurrencySymbol() {
        given()
            .when()
                .get("/v1/currency/EUR")
            .then()
                .statusCode(200)
                .body("currencies[0].symbol", equalTo("€"));
    }

    @Test(timeout = 60000)
    public void testV2RegionalblocReturnsCurrencyFields() {
        given()
            .when()
                .queryParam("fields", "name;capital;currencies")
                .get("/v2/regionalbloc/EU")
            .then()
                .statusCode(200)
                .body("currencies[0][0].code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1NameReturnsCurrencyCodeAndName() {
        given()
            .when()
                .get("/v1/name/France")
            .then()
                .statusCode(200)
                .body("currencies[0].code", equalTo("EUR"));
    }

    @Test(timeout = 60000)
    public void testV1AllReturnsCurrencySymbol() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .statusCode(200)
                .body("currencies[0][0].symbol", notNullValue());
    }
}
