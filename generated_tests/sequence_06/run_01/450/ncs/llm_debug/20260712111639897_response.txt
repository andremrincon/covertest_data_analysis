package ts01gpt_5_mini;

import io.restassured.response.Response;
import org.junit.Test;
import java.util.Optional;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class RemainderTest {

    private static final String BASE = Optional.ofNullable(System.getProperty("api.base"))
            .orElse(Optional.ofNullable(System.getenv("API_BASE")).orElse("http://localhost:8080"));

    @Test(timeout = 60000)
    public void testAEqualsZeroReturnsMinusOne() {
        given().when().get(BASE + "/api/triangle/1/1/1").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/0/5");
        act.then().body("result", equalTo(-1));
    }

    @Test(timeout = 60000)
    public void testBEqualsZeroReturnsMinusOne() {
        given().when().get(BASE + "/api/triangle/1/1/1").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/7/0");
        act.then().body("result", equalTo(-1));
    }

    @Test(timeout = 60000)
    public void testAPositiveBPositiveComputesRemainder() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/17/5");
        act.then().body("result", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testAPositiveBNegativeComputesRemainder() {
        given().when().get(BASE + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/17/-9");
        act.then().body("result", equalTo(8));
    }

    @Test(timeout = 60000)
    public void testANegativeBPositiveComputesRemainder() {
        given().when().get(BASE + "/api/bessj/3/2.5").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/-17/5");
        act.then().body("result", equalTo(-2));
    }

    @Test(timeout = 60000)
    public void testANegativeBNegativeComputesRemainder() {
        given().when().get(BASE + "/api/bessj/3/2.5").then().statusCode(lessThan(300));
        Response act = given().when().get(BASE + "/api/remainder/-17/-9");
        act.then().body("result", equalTo(8));
    }
}
