    @Test(timeout = 60000)
    public void testSetNameWhenCreatingFeature() {
        String productName = "Product-SetName-" + UUID.randomUUID().toString().substring(0, 8);
        given().when().post("/products/" + productName).then().statusCode(lessThan(300));

        String featureName = "Feature-SetName-" + UUID.randomUUID().toString().substring(0, 8);
        Response response = given()
                .formParam("description", "Test description for setName")
                .when()
                .post("/products/" + productName + "/features/" + featureName);

        response.then().statusCode(201);
    }
