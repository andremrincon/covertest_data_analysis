package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    @Before
    public void setup() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testCurrencyCodeAlpha() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testCurrencyNameAlpha() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().body("currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testCurrencySymbolAlpha() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().body("currencies[0].symbol", equalTo("$"));
    }

    @Test(timeout = 60000)
    public void testCurrencyCodeCurrency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().body("[0].currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testCurrencyNameCurrency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().body("[0].currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testCurrencySymbolCurrency() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().body("[0].currencies[0].symbol", equalTo("$"));
    }
}
