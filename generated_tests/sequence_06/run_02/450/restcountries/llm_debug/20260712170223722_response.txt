package ts01gpt_5_mini;

import org.junit.Test;
import io.restassured.response.Response;
import static org.hamcrest.Matchers.lessThan;
import static io.restassured.RestAssured.given;

public class CountryServiceBaseTest {

    private static final String BASE;
    static {
        String b = System.getProperty("api.base");
        if (b == null || b.isEmpty()) b = System.getenv("API_BASE");
        if (b == null || b.isEmpty()) b = "http://localhost:8080/rest";
        BASE = b;
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_TwoLetterReturns200() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().when().get(BASE + "/v1/alpha/US").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_InvalidFormatReturns400() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().when().get(BASE + "/v1/alpha/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_NotFoundReturns404() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().when().get(BASE + "/v1/alpha/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByCodeList_CSV_Returns200() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA,MX").when().get(BASE + "/v1/alpha").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCodeList_NotFound_Returns404() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "XX,YY,ZZ").when().get(BASE + "/v1/alpha").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testFulltextSearch_FullTextExactMatch_Returns200() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fullText", "true").when().get(BASE + "/v1/name/France").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearch_AlternativeSpelling_Returns200() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fullText", "true").when().get(BASE + "/v1/name/Deutschland").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testLoadJson_BadCodes_LeadsTo500() {
        given().when().get(BASE + "/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "[\"US\",\"CA\"]").when().get(BASE + "/v1/alpha").then().statusCode(500);
    }
}
