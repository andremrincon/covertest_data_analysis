    public static void setUp() {
        RestAssured.baseURI = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080");
        RestAssured.basePath = "/rest";
    }

    @Test(timeout = 60000)
    public void testGetAllCountriesV2() {
        given()
            .when()
                .get("/v2/all")
