package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class Text2TxtTest {
    
    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }
    
    @Test(timeout = 60000)
    public void testTwoReturnsAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/two/hello/world")
            .then()
            .statusCode(200);
    }
    
    @Test(timeout = 60000)
    public void testFourReturnsAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/four/hello/world")
            .then()
            .statusCode(200);
    }
    
    @Test(timeout = 60000)
    public void testAreReturnsAbbreviation() {
        given()
            .when()
            .get("/api/text2txt/are/hello/world")
            .then()
            .statusCode(200);
    }
    
    @Test(timeout = 60000)
    public void testSeeYouReturnsCu() {
        given()
            .when()
            .get("/api/text2txt/see/you/later")
            .then()
            .statusCode(200);
    }
    
    @Test(timeout = 60000)
    public void testByTheWayReturnsBtw() {
        given()
            .when()
            .get("/api/text2txt/by/the/way")
            .then()
            .statusCode(200);
    }
    
    @Test(timeout = 60000)
    public void testNoMatchReturnsDefault() {
        given()
            .when()
            .get("/api/text2txt/hello/world/test")
            .then()
            .statusCode(200);
    }
}
