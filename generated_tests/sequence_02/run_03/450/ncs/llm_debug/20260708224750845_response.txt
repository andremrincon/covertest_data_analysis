    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testBessj_N_LessThanOrEqualTo2_Returns400() {
        given()
            .when()
                .get("/api/bessj/2/2.5")
            .then()
                .statusCode(400);
    }
