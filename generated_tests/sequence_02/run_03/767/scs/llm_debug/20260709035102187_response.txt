package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class TitleTest {

    @Before
    public void setUp() {
        String host = System.getProperty("server.host", "localhost");
        String port = System.getProperty("server.port", "8080");
        RestAssured.baseURI = "http://" + host + ":" + port;
    }

    @Test(timeout = 60000)
    public void testSubjectTwo() {
        given()
                .when()
                .get("/api/text2txt/two/quick/brown")
                .then()
                .statusCode(200)
                .body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testSubjectFor() {
        given()
                .when()
                .get("/api/text2txt/for/quick/brown")

    }

    @Test(timeout = 60000)
    public void testSubjectFour() {

    }

    @Test(timeout = 60000)
    public void testSubjectYou() {

                .get("/api/text2txt/you/quick/brown")
                .then()
                .statusCode(200)
                .body(equalTo("u"));
    }

    @Test(timeout = 60000)

    }
}
