package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.notNullValue;

public class CurrencyTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", System.getenv().getOrDefault("BASE_URL", "http://localhost:8080/rest"));
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US").then().statusCode(200).body("currencies.USD", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2AlphaMultipleCodesReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA,GB").when().get("/v2/alpha").then().statusCode(200).body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/currency/EUR").then().statusCode(200).body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2NameEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/name/Germany").then().statusCode(200).body("currencies.EUR", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CallingCodeEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/callingcode/1").then().statusCode(200).body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CapitalEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/capital/Paris").then().statusCode(200).body("currencies", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2RegionEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fields", "name;capital;currencies").when().get("/v2/region/Europe").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV2RegionalblocEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fields", "name;capital;currencies").when().get("/v2/regionalbloc/EU").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV2AllEndpointReturnsCurrencyData() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fields", "name;currencies").when().get("/v2/all").then().statusCode(200);
    }
}
