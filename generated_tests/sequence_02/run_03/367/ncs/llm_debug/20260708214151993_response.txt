package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.response.Response;
import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class FisherTest {

    private static String BASE;

    @BeforeClass
    public static void setup() {
        String env = System.getenv("API_BASE_URL");
        if (env != null && !env.isEmpty()) {
            BASE = env;
        } else {
            String prop = System.getProperty("api.base");
            if (prop != null && !prop.isEmpty()) {
                BASE = prop;
            } else {
                BASE = "http://localhost:8080";
            }
        }
        RestAssured.baseURI = BASE;
    }

    @Test(timeout = 60000)
    public void testFisher_when_m1_n1_x075_returns200() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/fisher/1/1/0.75");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_when_m3_n4_x075_returns_success_message() {
        given().when().get("/api/expint/3/0.1").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/fisher/3/4/0.75");
        resp.then().body("message", equalTo("Fisher computation successful"));
    }

    @Test(timeout = 60000)
    public void testFisher_when_m10_n1_x0_returns200() {
        given().when().get("/api/remainder/17/5").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/fisher/10/1/0.0");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_when_m10_n6_x075_returns200() {
        given().when().get("/api/gammq/5.5/2.3").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/fisher/10/6/0.75");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_when_m2_n4_x_very_small_returns200() {
        given().when().get("/api/bessj/3/2.5").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/fisher/2/4/1e-10");
        resp.then().statusCode(200);
    }
}
