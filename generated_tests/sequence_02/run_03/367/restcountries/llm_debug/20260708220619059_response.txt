package ts01qwen3_7_plus;

import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CurrencyTest {

    private String baseUrl;

    @Before
    public void setUp() {
        String envUrl = System.getenv("BASE_URL");
        baseUrl = System.getProperty("baseUrl", envUrl != null ? envUrl : "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testSetCode() {
        given()
            .when()
                .get(baseUrl + "/v1/alpha/US")
            .then()
                .body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testSetName() {
        given()
            .when()
                .get(baseUrl + "/v1/alpha/US")
            .then()
                .body("currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testSetSymbol() {
        given()
            .when()
                .get(baseUrl + "/v1/alpha/US")
            .then()
                .body("currencies[0].symbol", equalTo("$"));
    }
}
