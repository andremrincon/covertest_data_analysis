    public static void setUp() {

        RestAssured.baseURI = baseUrl;

    @Test(timeout = 60000)
    public void testGetCountryByAlphaCodeReturnsLanguageFields() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body("languages[0].iso639_1", notNullValue())
                .body("languages[0].iso639_2", notNullValue())
