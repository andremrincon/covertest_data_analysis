package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryServiceBaseTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base", System.getenv("API_BASE_URL"));
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testV1Alpha2Success() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1Alpha3ViaCodesSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha?codes=USA,US");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1AlphaBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/123");
        resp.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV1AlphaNotFound() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/XYZ");
        resp.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha?codes=US,CA,MX");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesNotFound() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha?codes=XX,YY,ZZ");
        resp.then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV1NameFullTextExact() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/France?fullText=true");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1NameFullTextAlternative() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/name/United%20States%20of%20America?fullText=true");
        resp.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodesServerErrorLoadJson() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha?codes=%5B%22US%22,%20%22CA%22%5D");
        resp.then().statusCode(500);
    }
}
