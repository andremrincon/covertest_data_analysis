package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class RegexTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testSubjectReturnsUrlForValidUrlInput() {
        given()
            .baseUri(BASE_URL)
            .pathParam("txt", "http://abc/def")
        .when()
            .get("/api/pat/{txt}")
        .then()
            .statusCode(200)
            .body(equalTo("url"));
    }

    @Test(timeout = 60000)
    public void testSubjectReturnsDateForValidDateInput() {
        given()
            .baseUri(BASE_URL)
            .pathParam("txt", "mon01jan")
        .when()
            .get("/api/pat/{txt}")
        .then()
            .statusCode(200)
            .body(equalTo("date"));
    }

    @Test(timeout = 60000)
    public void testSubjectReturnsFpeForValidFpeInput() {
        given()
            .baseUri(BASE_URL)
            .pathParam("txt", "12.34e+56")
        .when()
            .get("/api/pat/{txt}")
        .then()
            .statusCode(200)
            .body(equalTo("fpe"));
    }

    @Test(timeout = 60000)
    public void testSubjectReturnsNoneForNonMatchingInput() {
        given()
            .baseUri(BASE_URL)
            .pathParam("txt", "hello")
        .when()
            .get("/api/pat/{txt}")
        .then()
            .statusCode(200)
            .body(equalTo("none"));
    }
}
