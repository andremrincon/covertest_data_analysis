package ts01gpt_5_mini;

import org.junit.Test;
import org.junit.BeforeClass;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class RegexTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testPatEndpointRecognizesUrlInput() {
        given().when().get("/api/text2txt/{w}/{x}/{y}", "The", "quick", "brown").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "http://a/b");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testPatEndpointRecognizesDateInput() {
        given().when().get("/api/text2txt/{w}/{x}/{y}", "A", "B", "C").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "mon12jan");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testPatEndpointRecognizesFloatingPointWithExponentInput() {
        given().when().get("/api/text2txt/{w}/{x}/{y}", "one", "two", "three").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "1.2e+34");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testPatEndpointReturnsNoneForNonMatchingInput() {
        given().when().get("/api/text2txt/{w}/{x}/{y}", "foo", "bar", "baz").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/pat/{txt}", "foobar");
        assertEquals(200, resp.getStatusCode());
    }
}
