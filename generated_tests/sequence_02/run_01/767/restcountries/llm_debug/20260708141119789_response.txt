package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryServiceBaseTest {

    private static final String BASE_URL = "http://localhost:8080/rest";

    @Test(timeout = 60000)
    public void testGetByAlphaTwoLetterCode() {
        given()
            .baseUri(BASE_URL)
            .pathParam("alphacode", "US")
        .when()
            .get("/v1/alpha/{alphacode}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCodeListValidCodes() {
        given()
            .baseUri(BASE_URL)
            .queryParam("codes", "US,CA")
        .when()
            .get("/v1/alpha")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearchAltSpelling() {
        given()
            .baseUri(BASE_URL)
            .pathParam("name", "United States of America")
            .queryParam("fullText", true)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubstringSearchAltSpelling() {
        given()
            .baseUri(BASE_URL)
            .pathParam("name", "Bundesrepublik")
            .queryParam("fullText", false)
        .when()
            .get("/v1/name/{name}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testLoadJson() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/v1/all")
        .then()
            .statusCode(200);
    }
}
