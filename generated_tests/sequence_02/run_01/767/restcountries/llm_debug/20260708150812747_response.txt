package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class LanguageTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("rest.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("REST_BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testV1Alpha_US_status200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1Name_France_languagesIso6391() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/name/France").then().body("[0].languages[0]['iso639_1']", equalTo((String) null));
    }

    @Test(timeout = 60000)
    public void testV1Currency_USD_status200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV1Lang_es_firstCountryLanguage() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/lang/es").then().body("[0].languages[0]['iso639_1']", equalTo((String) null));
    }

    @Test(timeout = 60000)
    public void testV2Alpha_US_withFields_status200() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US?fields=name;capital;population").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testV2Lang_Spanish_languageIso() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().when().get("/v2/lang/Spanish").then().body("[0].languages[0]['iso639_1']", equalTo((String) null));
    }
}
