package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import java.io.UnsupportedEncodingException;
import java.net.URLEncoder;
import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    private String encodePathSegment(String segment) {
        try {
            return URLEncoder.encode(segment, "UTF-8").replace("+", "%20");
        } catch (UnsupportedEncodingException e) {
            return segment.replace(" ", "%20");
        }
    }

    @Test(timeout = 60000)
    public void testGetAllProducts() {
        given()
            .when()
                .get("/products")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testOptionsRequest() {
        given()
            .when()
                .options("/products")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testAddProduct() {
        String productName = "Wireless Mouse " + UUID.randomUUID();

        given()
            .when()
                .post("/products/" + encodePathSegment(productName))
            .then()
                .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForProduct() {
        String productName = "AeroBook-Pro-15-" + UUID.randomUUID();
        given().when().post("/products/" + encodePathSegment(productName)).then().statusCode(lessThan(300));

        given()
            .when()
                .get("/products/" + encodePathSegment(productName) + "/features")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDeleteProduct() {
        String productName = "Smartwatch Series 9-" + UUID.randomUUID();
        given().when().post("/products/" + encodePathSegment(productName)).then().statusCode(lessThan(300));

        given()
            .when()
                .delete("/products/" + encodePathSegment(productName))
            .then()
                .statusCode(204);
    }
}
