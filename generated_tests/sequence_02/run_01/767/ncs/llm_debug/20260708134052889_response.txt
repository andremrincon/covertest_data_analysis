package ts01gpt_5_mini;

import org.junit.Test;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class RemainderTest {

    private String baseUrl() {
        String p = System.getProperty("api.base");
        if (p != null && !p.isEmpty()) return p;
        String e = System.getenv("API_BASE");
        if (e != null && !e.isEmpty()) return e;
        return "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testPositiveAPositiveB() {
        String base = baseUrl();
        given().when().get(base + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(base + "/api/remainder/17/5").then().statusCode(200).extract().response();
        String body = resp.getBody().asString().trim();
        int actual;
        if (body.startsWith("{") || body.startsWith("[")) {
            try {
                actual = resp.jsonPath().getInt("result");
            } catch (Exception e) {
                actual = Integer.parseInt(body.replaceAll("[^\\-0-9]", ""));
            }
        } else {
            actual = Integer.parseInt(body);
        }
        assertEquals(2, actual);
    }

    @Test(timeout = 60000)
    public void testPositiveANegativeB() {
        String base = baseUrl();
        given().when().get(base + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(base + "/api/remainder/17/-9").then().statusCode(200).extract().response();

    }

    @Test(timeout = 60000)
    public void testAZeroBadRequest() {
        String base = baseUrl();
        given().when().get(base + "/api/triangle/3/4/5").then().statusCode(lessThan(300));
        Response resp = given().when().get(base + "/api/remainder/0/5").then().statusCode(200).extract().response();
