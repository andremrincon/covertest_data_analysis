package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.hasItem;
import static org.hamcrest.Matchers.hasSize;
import static org.hamcrest.Matchers.lessThan;

public class ProductsConfigurationsServiceTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        RestAssured.baseURI = (baseUrl != null) ? baseUrl : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testGetConfigurationsNamesForProduct_NoConfigurations() {
        String productName = "Product-" + UUID.randomUUID().toString();

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/products/" + productName + "/configurations")
            .then()
            .body("$", hasSize(0));
    }

    @Test(timeout = 60000)
    public void testGetConfigurationsNamesForProduct_WithConfigurations() {
        String productName = "Product-" + UUID.randomUUID().toString();
        String configurationName = "Config-" + UUID.randomUUID().toString();

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/configurations/" + configurationName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/products/" + productName + "/configurations")
            .then()
            .body("$", hasItem(configurationName));
    }

    @Test(timeout = 60000)
    public void testEvaluateAndUpdateConfiguration_InvalidResult() {
        String productName = "Product-" + UUID.randomUUID().toString();
        String featureNameA = "FeatureA-" + UUID.randomUUID().toString();
        String featureNameB = "FeatureB-" + UUID.randomUUID().toString();
        String configurationName = "Config-" + UUID.randomUUID().toString();

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/features/" + featureNameA)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/features/" + featureNameB)
            .then()
            .statusCode(lessThan(300));

        given()
            .contentType("application/x-www-form-urlencoded")
            .formParam("sourceFeature", featureNameA)
            .formParam("excludedFeature", featureNameB)
            .when()
            .post("/products/" + productName + "/constraints/excludes")
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/configurations/" + configurationName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/configurations/" + configurationName + "/features/" + featureNameA)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/configurations/" + configurationName + "/features/" + featureNameB)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .get("/products/" + productName + "/configurations/" + configurationName)
            .then()
            .body("valid", equalTo(false));
    }
}
