package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CalcTest {

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testConstants() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/e/0/0").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnaryOperators() {
        String[] ops = {"sqrt", "log", "sine", "cosine"};
        for (String op : ops) {
            given().when().get("/api/calc/" + op + "/10/0").then().statusCode(lessThan(300));
        }
        given().when().get("/api/calc/tangent/10/0").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testBinaryOperators() {
        String[] ops = {"plus", "subtract", "multiply"};
        for (String op : ops) {
            given().when().get("/api/calc/" + op + "/10/5").then().statusCode(lessThan(300));
        }
        given().when().get("/api/calc/divide/10/5").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testUnknownOperator() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/unknown/10/5").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testInvalidArguments() {
        given().when().get("/api/calc/plus/10/5").then().statusCode(lessThan(300));
        given().when().get("/api/calc/plus/10/twenty").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testDivideByZero() {
        given().when().get("/api/calc/divide/10/5").then().statusCode(lessThan(300));
        given().when().get("/api/calc/divide/100/0").then().statusCode(500);
    }
}
