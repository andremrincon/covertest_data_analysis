package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertThat;

public class CookieTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testUseridWithValidUserValue() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/userid/user123/any").then().extract().asString();
        assertThat(response, equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testUseridWithShortValue() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/userid/user1/any").then().extract().asString();
        assertThat(response, equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testUseridWithNonUserValue() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/userid/admin123/any").then().extract().asString();
        assertThat(response, equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testSessionWithValidAmAndAbcCom() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/session/am/abc.com").then().extract().asString();
        assertThat(response, equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testSessionWithInvalidAmOrAbcCom() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/session/pm/abc.com").then().extract().asString();
        assertThat(response, equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testOtherName() {
        given().when().get(BASE_URL + "/api/cookie/other/any/any").then().statusCode(lessThan(300));
        String response = given().when().get(BASE_URL + "/api/cookie/other/any/any").then().extract().asString();
        assertThat(response, equalTo("0"));
    }
}
