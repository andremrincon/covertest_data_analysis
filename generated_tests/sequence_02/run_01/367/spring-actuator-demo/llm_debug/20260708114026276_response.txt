    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");

        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void timeConsumingAPI_delayZero_usesRandomDelay() {
        given()
                .queryParam("delay", 0)
            .when()
                .get("/slowApi")
            .then()
                .statusCode(401);
    }

    @Test(timeout = 60000)
    public void timeConsumingAPI_explicitDelay_returnsResult() {
        given()
                .queryParam("delay", 1)
            .when()
                .get("/slowApi")
            .then()
                .statusCode(401);
    }

    @Test(timeout = 60000)
    public void timeConsumingAPI_invalidDelayType_returns500() {
        given()
                .queryParam("delay", "abc")
            .when()
                .get("/slowApi")
            .then()
                .statusCode(401);
    }

    @Test(timeout = 60000)
    public void timeConsumingAPI_negativeDelay_returns500() {
        given()
                .queryParam("delay", -1)
            .when()
                .get("/slowApi")
            .then()
                .statusCode(401);
