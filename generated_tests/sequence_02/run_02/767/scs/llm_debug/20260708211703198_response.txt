package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class TitleTest {

    @BeforeClass
    public static void setUp() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testMaleTitleMr_returns1() {
        String seed = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", seed).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/title/{sex}/{title}", "male", "mr");
        assertEquals("1", resp.getBody().asString().trim());
    }

    @Test(timeout = 60000)
    public void testFemaleTitleMiss_returns0() {
        String seed = UUID.randomUUID().toString();
        given().when().get("/api/text2txt/{w}/{x}/{y}", seed, "setup", "value").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/title/{sex}/{title}", "female", "miss");
        assertEquals("0", resp.getBody().asString().trim());
    }

    @Test(timeout = 60000)
    public void testNoneTitleDr_returns2() {
        String seed = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", seed).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/title/{sex}/{title}", "none", "dr");
        assertEquals("2", resp.getBody().asString().trim());
    }

    @Test(timeout = 60000)
    public void testMaleTitleUnrecognized_returnsMinus1() {
        String seed = UUID.randomUUID().toString();
        given().when().get("/api/ordered4/{w}/{x}/{z}/{y}", seed, "a", "b", "c").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/title/{sex}/{title}", "male", "queen");
        assertEquals("-1", resp.getBody().asString().trim());
    }

    @Test(timeout = 60000)
    public void testFemaleTitleDr_returns0() {
        String seed = UUID.randomUUID().toString();
        given().when().get("/api/filesuffix/{directory}/{file}", "/tmp", seed).then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/title/{sex}/{title}", "female", "dr");
        assertEquals("0", resp.getBody().asString().trim());
    }
}
