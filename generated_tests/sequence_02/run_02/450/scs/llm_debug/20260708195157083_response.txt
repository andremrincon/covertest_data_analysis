package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class DateParseTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testValidDayAndMonthReturns200() {
        String[] months = {"jan","feb","mar","apr","may","jun","jul","aug","sep","oct","nov","dec"};
        for (String m : months) {
            given().when().get("/api/dateparse/{day}/{month}", "mon", m).then().statusCode(lessThan(300));
        }
        given().when().get("/api/dateparse/{day}/{month}", "Wednesday", "August").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testInvalidDayProduces500() {
        given().when().get("/api/dateparse/{day}/{month}", "mon", "jan").then().statusCode(lessThan(300));
        given().when().get("/api/dateparse/{day}/{month}", "123", "Movember").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testAlternateDaySpellingAndCaseInsensitivity() {
        String[] days = {"Mon","tue","WED","Thur","FRI","sat","SUN"};
        for (String d : days) {
            given().when().get("/api/dateparse/{day}/{month}", d, "MAR").then().statusCode(lessThan(300));
        }
        given().when().get("/api/dateparse/{day}/{month}", "thur", "Dec").then().statusCode(200);
    }
}
