package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class RegexTest {

    private String baseUrl;

    @Before
    public void setUp() {
        baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testFileSuffixNoExtension() {
        given()
            .when()
            .get("/api/filesuffix/unknown/file")
            .then()
            .statusCode(200)
            .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testFileSuffixTextTxt() {
        given()
            .when()
            .get("/api/filesuffix/text/file.txt")
            .then()
            .statusCode(200)
            .body(equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testFileSuffixAcrobatPdf() {
        given()
            .when()
            .get("/api/filesuffix/acrobat/file.pdf")
            .then()
            .statusCode(200)
            .body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testFileSuffixWordDoc() {
        given()
            .when()
            .get("/api/filesuffix/word/file.doc")
            .then()
            .statusCode(200)
            .body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testFileSuffixBinExe() {
        given()
            .when()
            .get("/api/filesuffix/bin/file.exe")
            .then()
            .statusCode(200)
            .body(equalTo("4"));
    }

    @Test(timeout = 60000)
    public void testFileSuffixLibDll() {
        given()
            .when()
            .get("/api/filesuffix/lib/file.dll")
            .then()
            .statusCode(200)
            .body(equalTo("5"));
    }
}
