package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class BessjTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void bessj_nLessThan2_returns400() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/1/2.5")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void bessj_zeroX_returns200() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/3/0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_negativeX_oddN_axGreaterThanN_returns200() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/3/-5.0")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_negativeX_largeAx_bessj1NegativeBranch_returns200() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/2/-8.5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_positiveX_axLessThanOrEqualN_returns200() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/3/2.5")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void bessj_positiveX_largeAx_bessj1PositiveBranch_returns200() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/api/bessj/2/8.5")
        .then()
            .statusCode(200);
    }
}
