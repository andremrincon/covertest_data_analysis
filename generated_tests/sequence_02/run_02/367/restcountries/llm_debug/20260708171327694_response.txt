package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class StripeRestTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost";
        RestAssured.port = 8080;
        RestAssured.basePath = "/rest";
    }

    @Test(timeout = 60000)
    public void contributeWithNullTokenReturnsBadRequest() {
        String body = "{}";
        given()
            .contentType(ContentType.JSON)
            .body(body)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void contributeWithBlankTokenReturnsBadRequest() {
        String body = "{\"amount\":1000,\"token\":\"   \"}";
        given()
            .contentType(ContentType.JSON)
            .body(body)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void contributeWithValidTokenReturnsBadRequestDueToStripeException() {
        String body = "{\"amount\":1000,\"token\":\"tok_test_12345\"}";
        given()
            .contentType(ContentType.JSON)
            .body(body)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }
}
