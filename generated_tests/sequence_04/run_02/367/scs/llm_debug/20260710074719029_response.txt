package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class NotyPevarTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testNotyPevar_i28_sWorld() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/notypevar/28/world")
        .then()
            .statusCode(200)
            .body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevar_i7_sAbc() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/notypevar/7/abc")
        .then()
            .statusCode(200)
            .body(equalTo("3"));
    }

    @Test(timeout = 60000)
    public void testNotyPevar_i3_sWorld() {
        given()
            .baseUri(BASE_URL)
        .when()
            .get("/api/notypevar/3/world")
        .then()
            .statusCode(200)
            .body(equalTo("2"));
    }
}
