package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class FisherTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) base = System.getenv("API_BASE");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testFisher_whenAEquals1AndBEquals1_returns200() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/1/1/0.75").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_whenAEquals1AndBNot1_returns200() {
        given().when().get("/api/remainder/17/5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/5/2/0.75").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_whenANot1AndBEquals1_returns200() {
        given().when().get("/api/expint/3/2.5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/4/3/0.75").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_whenANot1AndBNot1_returns200() {
        given().when().get("/api/bessj/3/2.5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/10/4/0.75").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisher_withInvalidM_returns400() {
        given().when().get("/api/triangle/3/4/5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/abc/1/0.75").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testFisher_withInvalidX_returns400() {
        given().when().get("/api/remainder/17/5").then().statusCode(lessThan(300));
        given().when().get("/api/fisher/1/1/1.2").then().statusCode(400);
    }
}
