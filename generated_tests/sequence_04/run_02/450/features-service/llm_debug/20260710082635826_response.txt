        return "Product-" + UUID.randomUUID().toString().substring(0, 8);

        return "Feature-" + UUID.randomUUID().toString().substring(0, 8);

        return "Config-" + UUID.randomUUID().toString().substring(0, 8);

    @Test(timeout = 60000)
    public void testAddFeatureToProduct() {
        String productName = uniqueProductName();
        String featureName = uniqueFeatureName();

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/features/" + featureName)
            .then()
            .statusCode(201);
    }
