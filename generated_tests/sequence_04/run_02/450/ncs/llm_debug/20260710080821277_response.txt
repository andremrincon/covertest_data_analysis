package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.junit.Assert.assertEquals;

public class ExpintTest {

    @Test(timeout = 60000)
    public void testExpintN0() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        
        Response response = given()
                .when()
                .get(baseUrl + "/api/expint/0/2.5");
        
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testExpintX0() {

        Response response = given()
                .when()
                .get(baseUrl + "/api/expint/2/0.0");
        
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testExpintXGreaterThan1() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        
        Response response = given()
                .when()
                .get(baseUrl + "/api/expint/3/2.5");
        
        assertEquals(200, response.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testExpintXLessThanOrEqualTo1() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        
        Response response = given()
                .when()

        assertEquals(200, response.getStatusCode());
    }
