package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CurrencyTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        RestAssured.baseURI = baseUrl != null ? baseUrl : "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeCurrencyCode() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeCurrencyName() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].name", equalTo("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaCodeCurrencySymbol() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("currencies[0].symbol", equalTo("$"));
    }

    @Test(timeout = 60000)
    public void testV2CurrencyCode() {
        given()
            .when()
                .get("/v2/currency/EUR")
            .then()
                .body("[0].currencies[0].code", equalTo("EUR"));
    }

    @Test(timeout = 60000)
    public void testV2CurrencyName() {
        given()
            .when()
                .get("/v2/currency/EUR")
            .then()
                .body("[0].currencies[0].name", equalTo("Euro"));
    }

    @Test(timeout = 60000)
    public void testV2CurrencySymbol() {
        given()
            .when()
                .get("/v2/currency/EUR")
            .then()
                .body("[0].currencies[0].symbol", equalTo("\u20ac"));
    }
}
