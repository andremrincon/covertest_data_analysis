package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryServiceBaseTest {

    private final String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080/rest";

    @Test(timeout = 60000)
    public void testGetByAlpha2Code() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/v2/alpha/US")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha3Code() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/v2/alpha/USA")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaNotFound() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/v2/alpha/ZZ")
        .then()
            .statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByCodeList() {
        given()
            .baseUri(baseUrl)
            .queryParam("codes", "US;CA")
        .when()
            .get("/v2/alpha")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCodeListDuplicates() {
        given()
            .baseUri(baseUrl)
            .queryParam("codes", "US;US")
        .when()
            .get("/v2/alpha")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearchExactName() {
        given()
            .baseUri(baseUrl)
            .queryParam("fullText", true)
        .when()
            .get("/v2/name/Germany")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFulltextSearchAltSpelling() {
        given()
            .baseUri(baseUrl)
            .queryParam("fullText", true)
        .when()
            .get("/v2/name/Federal Republic of Germany")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testLoadJson() {
        given()
            .baseUri(baseUrl)
        .when()
            .get("/v2/all")
        .then()
            .statusCode(200);
    }
}
