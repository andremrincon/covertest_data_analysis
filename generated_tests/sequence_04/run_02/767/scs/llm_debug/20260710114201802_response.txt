package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import io.restassured.response.Response;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class RegexTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSubjectRecognizesUrl() {
        String unique = UUID.randomUUID().toString().substring(0, 8);
        given().when().get("/api/text2txt/{w}/{x}/{y}", "The", "quick", unique).then().statusCode(lessThan(300));
        Response r = given().when().get("/api/pat/{txt}", "http://abc/def");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubjectRecognizesDate() {
        String unique = UUID.randomUUID().toString().substring(0, 8);
        given().when().get("/api/text2txt/{w}/{x}/{y}", "A", "b", unique).then().statusCode(lessThan(300));
        Response r = given().when().get("/api/pat/{txt}", "mon01jan");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubjectRecognizesFpe() {
        String unique = UUID.randomUUID().toString().substring(0, 8);
        given().when().get("/api/text2txt/{w}/{x}/{y}", "X", "Y", unique).then().statusCode(lessThan(300));
        Response r = given().when().get("/api/pat/{txt}", "1.2e+03");
        r.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubjectReturnsNoneForNonMatchingText() {
        String unique = UUID.randomUUID().toString().substring(0, 8);
        given().when().get("/api/text2txt/{w}/{x}/{y}", "foo", "bar", unique).then().statusCode(lessThan(300));
        Response r = given().when().get("/api/pat/{txt}", "randomXYZ123");
        r.then().statusCode(200);
    }
}
