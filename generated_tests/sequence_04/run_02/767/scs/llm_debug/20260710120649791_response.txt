package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CalcTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ?
                System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testCalcEndpointWithPi() {
        given().when().get("/api/calc/pi/0/0").then().statusCode(lessThan(300));

        given()
                .when()
                .get("/api/calc/pi/0/0")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCalcEndpointWithSqrt() {
        given().when().get("/api/calc/sqrt/16/0").then().statusCode(lessThan(300));

        given()
                .when()
                .get("/api/calc/sqrt/16/0")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCalcEndpointWithPlus() {
        given().when().get("/api/calc/plus/5/3").then().statusCode(lessThan(300));

        given()
                .when()
                .get("/api/calc/plus/5/3")
                .then()
                .statusCode(200);
    }
}
