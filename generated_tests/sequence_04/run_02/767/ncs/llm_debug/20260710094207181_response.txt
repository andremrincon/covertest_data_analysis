package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;

public class GammqTest {

    private String getBaseUrl() {
        String baseUrl = System.getenv("BASE_URL");
        return (baseUrl != null && !baseUrl.isEmpty()) ? baseUrl : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testGserXZero() {
        given()
            .pathParam("a", 5.5)
            .pathParam("x", 0.0)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserNormal() {
        given()
            .pathParam("a", 5.5)
            .pathParam("x", 2.3)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserTooLargeA() {
        given()
            .pathParam("a", 1000.0)
            .pathParam("x", 0.1)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGcfNormal() {
        given()
            .pathParam("a", 0.001)
            .pathParam("x", 1000.0)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGcfSmallD() {
        given()
            .pathParam("a", 0.001)
            .pathParam("x", 1e31)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGcfTooLargeA() {
        given()
            .pathParam("a", 1000.0)
            .pathParam("x", 2000.0)
        .when()
            .get(getBaseUrl() + "/api/gammq/{a}/{x}")
        .then()
            .statusCode(400);
    }
}
