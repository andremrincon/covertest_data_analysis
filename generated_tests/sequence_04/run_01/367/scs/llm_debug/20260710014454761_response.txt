package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.*;
import io.restassured.RestAssured;
import static org.hamcrest.Matchers.lessThan;

public class RegexTest {

    @BeforeClass
    public static void setup() {
        String base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testUrlClassification() {
        String dummy = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", dummy).then().statusCode(lessThan(300));
        String txt = "http://abc/def";
        given().when().get("/api/pat/{txt}", txt).then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testDateClassification() {
        String dummy = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", dummy).then().statusCode(lessThan(300));
        String txt = "mon12jan";
        given().when().get("/api/pat/{txt}", txt).then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFpeClassification() {
        String dummy = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", dummy).then().statusCode(lessThan(300));
        String txt = "12.3e+45";
        given().when().get("/api/pat/{txt}", txt).then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNoneClassification() {
        String dummy = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", dummy).then().statusCode(lessThan(300));
        String txt = "banana";
        given().when().get("/api/pat/{txt}", txt).then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatEndpointPatternSearch() {
        String arrangeId = UUID.randomUUID().toString();
        given().when().get("/api/pat/{txt}", arrangeId).then().statusCode(lessThan(300));
        String txt = "ABABCABAB";
        String pat = "ABAB";
        given().when().get("/api/pat/{txt}/{pat}", txt, pat).then().statusCode(200);
    }
}
