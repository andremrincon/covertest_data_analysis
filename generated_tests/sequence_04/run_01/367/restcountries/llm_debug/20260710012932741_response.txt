package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void doFilter_addsAccessControlAllowOriginHeader_onGetAll() {
        given()
                .when()
                .get("/v1/all")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsAccessControlAllowMethodsHeader_onGetByAlphaCode() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsAccessControlAllowHeadersHeader_onGetByName() {
        given()
                .when()
                .get("/v1/name/France")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCacheControlHeader_onGetByCapital() {
        given()
                .when()
                .get("/v1/capital/London")
                .then()
                .statusCode(404)
                .header("Cache-Control", equalTo("public, max-age=86400"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsAllCORSHeaders_onGetByRegion() {
        given()
                .when()
                .get("/v1/region/Europe")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"))
                .header("Access-Control-Allow-Methods", equalTo("GET"))
                .header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"))
                .header("Cache-Control", equalTo("public, max-age=86400"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onGetByCallingCode() {
        given()
                .when()
                .get("/v1/callingcode/1")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onGetByCurrency() {
        given()
                .when()
                .get("/v1/currency/USD")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onV2All() {
        given()
                .when()
                .get("/v2/all")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onV2Alpha() {
        given()
                .queryParam("codes", "US,CA,MX")
                .when()
                .get("/v2/alpha")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onV2Name() {
        given()
                .when()
                .get("/v2/name/Germany")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onV2Region() {
        given()
                .when()
                .get("/v2/region/Europe")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void doFilter_addsCORSHeaders_onV2Subregion() {
        given()
                .when()
                .get("/v2/subregion/{sub}", "Western Europe")
                .then()
                .statusCode(404)
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }
}
