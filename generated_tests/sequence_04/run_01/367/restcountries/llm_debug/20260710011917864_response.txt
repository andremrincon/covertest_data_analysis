package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import java.net.URL;
import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @BeforeClass
    public static void setUp() throws Exception {
        String base = System.getProperty("api.base");
        if (base == null) base = System.getenv("API_BASE");
        if (base == null) base = "http://localhost:8080/rest";
        try {
            URL u = new URL(base);
            RestAssured.baseURI = u.getProtocol() + "://" + u.getHost();
            if (u.getPort() != -1 && u.getPort() != 0) RestAssured.port = u.getPort();
            RestAssured.basePath = u.getPath();
        } catch (Exception e) {
            RestAssured.baseURI = base;
        }
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowOriginOnAlphaUS() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().get("/v1/alpha/US").then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowMethodsHeaderOnV2All() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().get("/v2/all").then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowHeadersOnNameFrance() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        when().get("/v1/name/France").then().header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnRegionEurope() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().get("/v1/region/Europe").then().header("Cache-Control", "public, max-age=86400");
    }

    @Test(timeout = 60000)
    public void testPostContributeReturnsAccepted() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().contentType(ContentType.JSON).body("{\"amount\":100,\"currency\":\"USD\",\"token\":\"tok_visa\"}")
                .when().post("/contribute").then().statusCode(202);
    }

    @Test(timeout = 60000)
    public void testInvalidAlphaReturnsBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().get("/v1/alpha/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testAlphaNotFoundReturns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().get("/v1/alpha/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testOptionsRequestHasCorsMethodsHeader() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        when().options("/v1/all").then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testAlphaMultipleCodesHasCorsOrigin() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA").when().get("/v1/alpha").then().header("Access-Control-Allow-Origin", "*");
    }
}
