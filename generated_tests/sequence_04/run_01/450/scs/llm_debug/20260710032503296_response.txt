package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CalcTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testPiConstantReturnsPiString() {
        given().when().get("/api/calc/e/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sqrt/4/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/log/2.718281828459045/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sine/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/cosine/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/tangent/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/plus/1/2").then().statusCode(lessThan(300));
        given().when().get("/api/calc/subtract/5/3").then().statusCode(lessThan(300));
        given().when().get("/api/calc/multiply/2/3").then().statusCode(lessThan(300));
        given().when().get("/api/calc/divide/6/2").then().statusCode(lessThan(300));
        given().when().get("/api/calc/pi/0/0").then().assertThat().body(equalTo(String.valueOf(Math.PI)));
    }

    @Test(timeout = 60000)
    public void testEConstantReturnsEString() {
        given().when().get("/api/pat/healthcheck").then().statusCode(lessThan(300));
        given().when().get("/api/calc/e/0/0").then().assertThat().body(equalTo(String.valueOf(Math.E)));
    }

    @Test(timeout = 60000)
    public void testSqrtReturnsSquareRoot() {
        given().when().get("/api/calc/plus/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sqrt/9/0").then().assertThat().body(equalTo(String.valueOf(Math.sqrt(9.0))));
    }

    @Test(timeout = 60000)
    public void testLogReturnsNaturalLog() {
        given().when().get("/api/calc/sqrt/16/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/log/1/0").then().assertThat().body(equalTo(String.valueOf(Math.log(1.0))));
    }

    @Test(timeout = 60000)
    public void testSineAtPiOverTwoReturnsOne() {
        given().when().get("/api/calc/cosine/0/0").then().statusCode(lessThan(300));
        given().when().get("/api/calc/sine/" + String.valueOf(Math.PI / 2.0) + "/0").then().assertThat().body(equalTo(String.valueOf(Math.sin(Math.PI / 2.0))));
    }

    @Test(timeout = 60000)
    public void testDivideByZeroReturnsServerError() {
        given().when().get("/api/calc/divide/10/2").then().statusCode(lessThan(300));
        given().when().get("/api/calc/divide/100/0").then().statusCode(500);
    }
}
