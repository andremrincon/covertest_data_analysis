package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;

public class CostfunsTest {

    private String getBaseUrl() {
        String envUrl = System.getenv("BASE_URL");
        return envUrl != null ? envUrl : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testCostfunsBranchIEquals5() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/5/test")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsBranchILessThanMinus444() {
        given()
            .baseUri(getBaseUrl())
        .when()
            .get("/api/costfuns/-500/test")
        .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testCostfunsBranchIGreaterThan666() {
        given()
            .baseUri(getBaseUrl())
