package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class GammqTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testGammqGcfNormalExecution() {
        given()
            .when()
            .get("/api/gammq/5.5/1000.0")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqGcfLargeA() {
        given()
            .when()
            .get("/api/gammq/10000.0/20000.0")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqGserNormalExecution() {
        given()
            .when()
            .get("/api/gammq/5.5/2.3")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqGserLargeA() {
        given()
            .when()
            .get("/api/gammq/10000.0/5000.0")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqXZero() {
        given()
            .when()
            .get("/api/gammq/5.5/0.0")
            .then()
            .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqInvalidA() {
        given()
            .when()
            .get("/api/gammq/-1.0/3.0")
            .then()
            .statusCode(400);
    }
}
