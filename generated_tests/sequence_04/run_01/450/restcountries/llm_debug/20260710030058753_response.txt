package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.Optional;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class LanguageTest {

    @BeforeClass
    public static void init() {
        String base = System.getProperty("api.base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("API_BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testV1AlphaUS_languagesIso639_1Set() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().body("languages[0].iso639_1", equalTo("en"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaBadFormatReturns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV1AlphaNotFoundReturns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testV2AlphaUS_nameSet() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().when().get("/v2/alpha/US").then().body("name", equalTo("United States"));
    }

    @Test(timeout = 60000)
    public void testV1Lang_es_languageIso639_1Present() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/lang/es").then().body("[0].languages[0].iso639_1", equalTo("es"));
    }

    @Test(timeout = 60000)
    public void testV2Lang_Spanish_languageIso639_1Present() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        given().when().get("/v2/lang/Spanish").then().body("[0].languages[0].iso639_1", equalTo("es"));
    }

    @Test(timeout = 60000)
    public void testV1Currency_USD_currencyCodeSet() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().body("[0].currencies[0].code", equalTo("USD"));
    }

    @Test(timeout = 60000)
    public void testV1CurrencyBadFormatReturns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testV1Name_France_returnsNameInBody() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/name/France").then().body("[0].name", equalTo("France"));
    }
}
