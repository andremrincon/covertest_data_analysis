    @Test(timeout = 60000)
    public void testSlowApiWithExplicitDelay() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 1).when().get("/slowApi").then().extract().response();
        assertEquals(401, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testSlowApiWithZeroDelayRandomBranch() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 0).when().get("/slowApi").then().extract().response();
        assertEquals(401, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testSlowApiWithNegativeDelayReturnsServerError() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", -1).when().get("/slowApi").then().extract().response();
        assertEquals(401, resp.getStatusCode());
    }
