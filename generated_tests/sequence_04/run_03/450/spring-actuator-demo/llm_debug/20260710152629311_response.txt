package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class SampleControllerTest {

    @BeforeClass
    public static void setup() {
        String envUrl = System.getProperty("base.url");
        if (envUrl == null || envUrl.isEmpty()) {
            envUrl = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080");
        }
        RestAssured.baseURI = envUrl;
    }

    @Test(timeout = 60000)
    public void testSayHelloWithName() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("name", "John Smith").when().get("/").then().extract().response();
        assertEquals("Hello John Smith!!", resp.asString());
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultName() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().when().get("/").then().extract().response();
        assertEquals("Hello Guest!!", resp.asString());
    }

    @Test(timeout = 60000)
    public void testSlowApiWithExplicitDelay() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 1).when().get("/slowApi").then().extract().response();
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testSlowApiWithZeroDelayRandomBranch() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", 0).when().get("/slowApi").then().extract().response();
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testSlowApiWithNegativeDelayReturnsServerError() {
        given().when().get("/").then().statusCode(lessThan(300));
        Response resp = given().queryParam("delay", -1).when().get("/slowApi").then().extract().response();
        assertEquals(500, resp.getStatusCode());
    }
}
