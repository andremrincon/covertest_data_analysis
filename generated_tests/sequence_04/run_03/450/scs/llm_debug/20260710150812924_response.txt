package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;

public class NotyPevarTest {
    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = System.getenv().getOrDefault("APP_BASE_URL", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testSubjectI0TrueI2TrueI3True() {
        given().when().get("/api/notypevar/28/zzz").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubjectI1TrueI3True() {
        given().when().get("/api/notypevar/7/a").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testSubjectAllFalse() {
        given().when().get("/api/notypevar/0/a").then().statusCode(200);
    }
}
