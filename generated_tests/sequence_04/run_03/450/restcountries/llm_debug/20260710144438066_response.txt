package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.nullValue;

public class CORSFilterTest {
    @BeforeClass
    public static void initBaseUri() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowOriginOnV1All() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        given().when().get("/v1/all").then().statusCode(200).header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowMethodsOnAlphaUS() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().statusCode(200).header("Access-Control-Allow-Methods", nullValue());
    }

    @Test(timeout = 60000)
    public void testAccessControlAllowHeadersOnAlphaInvalidNumeric() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/123").then().statusCode(200).header("Access-Control-Allow-Headers", nullValue());
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnAlphaNotFound() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/XYZ").then().statusCode(200).header("Cache-Control", nullValue());
    }

    @Test(timeout = 60000)
    public void testCORSHeadersOnV2AlphaWithCodesParam() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().param("codes", "US,CA").when().get("/v2/alpha").then().statusCode(200).header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void testCORSOnCapitalEndpointReturnsAllowMethods() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v2/capital/Paris").then().statusCode(200).header("Access-Control-Allow-Methods", nullValue());
    }

    @Test(timeout = 60000)
    public void testCORSOnNameEndpointWithFullTextTrue() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().param("fullText", "true").when().get("/v1/name/France").then().statusCode(200).header("Access-Control-Allow-Origin", nullValue());
    }

    @Test(timeout = 60000)
    public void testPostContributeCacheControlHeader() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().contentType("application/json").body("{}").when().post("/contribute").then().statusCode(200).header("Cache-Control", nullValue());
    }
}
