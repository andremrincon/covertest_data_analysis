package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryRestV2Test {

    @Test(timeout = 60000)
    public void testAllEndpoints_200() {
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/US").then().statusCode(200);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/?codes=US,CA,MX").then().statusCode(400);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/currency/EUR").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/name/Germany").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/callingcode/1").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/capital/Paris").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/region/Europe").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/subregion/Western%20Europe").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/lang/Spanish").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/demonym/American").then().statusCode(lessThan(300));
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/regionalbloc/EU").then().statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testAllEndpoints_400() {
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/?codes=[\"US\", \"CA\"]").then().statusCode(400);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/currency/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testAllEndpoints_404() {
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/ZZZ").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/?codes=XX,YY,ZZ").then().statusCode(400);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/currency/XYZ").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/name/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/callingcode/abc").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/capital/12345").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/region/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/subregion/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/lang/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/demonym/123").then().statusCode(404);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/regionalbloc/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testAllEndpoints_500() {
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/alpha/?codes=US%7CCA%7CMX").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/currency/{}").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/name/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/callingcode/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/capital/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/region/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/subregion/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/lang/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/demonym/True").then().statusCode(500);
        given().baseUri("http://localhost:8080").basePath("/rest").when().get("/v2/regionalbloc/True").then().statusCode(500);
    }
}
