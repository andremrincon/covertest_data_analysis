package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class ContributionTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testContributeAcceptedWithValidContribution() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        String body = "{\"amount\":500,\"currency\":\"EUR\",\"token\":\"tok_visa\",\"reference\":\"" + UUID.randomUUID().toString() + "\"}";
        Response res = given().contentType(ContentType.JSON).body(body).when().post("/contribute");
        assertEquals(202, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestWhenMissingToken() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        String body = "{\"amount\":100,\"currency\":\"EUR\",\"reference\":\"" + UUID.randomUUID().toString() + "\"}";
        Response res = given().contentType(ContentType.JSON).body(body).when().post("/contribute");
        assertEquals(400, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestWhenMissingAmount() {
        given().when().get("/v1/alpha?codes=US").then().statusCode(lessThan(300));
        String body = "{\"token\":\"tok_visa\",\"currency\":\"EUR\",\"reference\":\"" + UUID.randomUUID().toString() + "\"}";
        Response res = given().contentType(ContentType.JSON).body(body).when().post("/contribute");
        assertEquals(400, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testContributeBadRequestWhenEmptyToken() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String body = "{\"amount\":50,\"currency\":\"USD\",\"token\":\"\",\"reference\":\"" + UUID.randomUUID().toString() + "\"}";
        Response res = given().contentType(ContentType.JSON).body(body).when().post("/contribute");
        assertEquals(400, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetAlphaWithMultipleCodesReturns200() {
        given().when().get("/v1/alpha?codes=US,CA").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha?codes=US,CA");
        assertEquals(200, res.getStatusCode());
    }
}
