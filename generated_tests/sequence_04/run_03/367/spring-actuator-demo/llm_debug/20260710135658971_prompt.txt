import static org.hamcrest.Matchers.lessThan;

public class SampleControllerTest {

    private static String base;

    @BeforeClass
    public static void setup() {
        String env = System.getProperty("baseUrl");
        if (env == null || env.isEmpty()) {
            env = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080");
        }
        base = env;
    }

    @Test(timeout = 60000)
    public void testSayHelloDefaultNameReturnsGuestGreeting() throws Exception {
        given().when().get(base + "/").then().statusCode(lessThan(300));
        Response act = given().when().get(base + "/");
        assertEquals("Hello Guest!!", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testSayHelloWithSpecialCharactersInNameReturnsGreeting() throws Exception {
        given().when().get(base + "/").then().statusCode(lessThan(300));
        String name = "María-José O'Connor-Smith III";
        String encoded = URLEncoder.encode(name, "UTF-8");
        Response act = given().when().get(base + "/?name=" + encoded);
        assertEquals("Hello " + name + "!!", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithExplicitPositiveDelayReturns200() throws Exception {
        given().when().get(base + "/").then().statusCode(lessThan(300));
        Response act = given().when().get(base + "/slowApi?delay=1");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithZeroDelayUsesRandomPathAndReturns200() throws Exception {
        given().when().get(base + "/").then().statusCode(lessThan(300));

        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testTimeConsumingApiWithNegativeDelayProducesServerError500() throws Exception {

}
