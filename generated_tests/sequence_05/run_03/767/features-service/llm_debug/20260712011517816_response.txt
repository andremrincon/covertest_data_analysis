    @Test(timeout = 60000)
    public void testCreateRequiresConstraintExercisesSetIdAndSetForProduct() {
        String productName = "TestProd-Req-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "FeatureA-" + UUID.randomUUID().toString().substring(0, 8);
        String requiredFeature = "FeatureB-" + UUID.randomUUID().toString().substring(0, 8);

        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));

        given()
            .formParam("sourceFeature", sourceFeature)
            .formParam("requiredFeature", requiredFeature)
            .when()
            .post("/products/" + productName + "/constraints/requires")
            .then()
            .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testDeleteRequiresConstraintExercisesGetId() {
        String productName = "TestProd-DelReq-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "FeatI-" + UUID.randomUUID().toString().substring(0, 8);
        String requiredFeature = "FeatJ-" + UUID.randomUUID().toString().substring(0, 8);

        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/" + sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/" + requiredFeature).then().statusCode(lessThan(300));

        Response createResp = given()
            .formParam("sourceFeature", sourceFeature)
            .formParam("requiredFeature", requiredFeature)
            .when()
            .post("/products/" + productName + "/constraints/requires");
        createResp.then().statusCode(lessThan(300));

        Integer constraintId = createResp.jsonPath().get("id");
        if (constraintId == null) {
            constraintId = createResp.jsonPath().get("data.id");
        }

        given()
            .when()
            .delete("/products/" + productName + "/constraints/" + constraintId)
            .then()
            .statusCode(204);
    }
