package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class PatTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testPatternFoundReverseNotFound() {
        given()
            .pathParam("txt", "ABABC")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("1"));
    }

    @Test(timeout = 60000)
    public void testPatternAndReverseFoundAsPalindrome() {
        given()
            .pathParam("txt", "ABABBABA")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testPatternAndReverseFoundSeparately() {
        given()
            .pathParam("txt", "ABABxxBABA")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testReverseFoundPatternNotFound() {
        given()
            .pathParam("txt", "BABAxx")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("2"));
    }

    @Test(timeout = 60000)
    public void testReverseAndPatternFoundAsPalindrome() {
        given()
            .pathParam("txt", "BABAABAB")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testReverseAndPatternFoundSeparately() {
        given()
            .pathParam("txt", "BABAxxABAB")
            .pathParam("pat", "ABAB")
        .when()
            .get("/api/pat/{txt}/{pat}")
        .then()
            .statusCode(200)
            .body(equalTo("0"));
    }
}
