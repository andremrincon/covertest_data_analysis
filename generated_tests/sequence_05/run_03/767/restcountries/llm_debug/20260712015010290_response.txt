    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_InvalidLength() {
        given()
            .queryParam("fields", "name")
        .when()
            .get("/v2/alpha/1")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_Success() {
        given()
            .queryParam("codes", "US,CA")

        .then()
            .statusCode(200);
    }
