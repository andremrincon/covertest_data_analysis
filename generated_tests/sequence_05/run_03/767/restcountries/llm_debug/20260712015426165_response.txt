package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CORSFilterTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080/rest";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAllowOriginOnV1AlphaSuccess() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha/US");
        res.then().header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void testAllowMethodsHeaderPresentOnV1All() {
        given().when().get("/v1/alpha/US").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/all");
        res.then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testAllowHeadersPresentOnV1AlphaList() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha?codes=US,CA");
        res.then().header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void testCacheControlHeaderOnV1Region() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/region/Europe");
        res.then().header("Cache-Control", "public, max-age=86400");
    }

    @Test(timeout = 60000)
    public void testCORSOnOptionsRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().options("/v1/alpha/US");
        res.then().header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void testCORSOnErrorResponse() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().when().get("/v1/alpha/123");
        res.then().header("Access-Control-Allow-Origin", "*");
    }
}
