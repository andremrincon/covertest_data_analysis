package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    private static String base;

    @BeforeClass
    public static void init() {
        String prop = System.getProperty("TEST_BASE_URL");
        String env = System.getenv("TEST_BASE_URL");
        if (prop != null && !prop.isEmpty()) {
            base = prop;
        } else if (env != null && !env.isEmpty()) {
            base = env;
        } else {
            base = "http://localhost:8080/rest";
        }
    }

    @Test(timeout = 60000)
    public void contribute_withBlankToken_returnsBadRequest_1() {
        given().when().get(base + "/v2").then().statusCode(lessThan(300));
        String body = "{\"amount\":100,\"token\":\"\"}";
        given().contentType("application/json;charset=utf-8").body(body).when().post(base + "/contribute").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void contribute_withValidTestToken_returnsAccepted_2() {
        given().when().get(base + "/v2").then().statusCode(lessThan(300));
        String token = "tok_visa_" + UUID.randomUUID().toString();
        String body = "{\"amount\":1500,\"token\":\"" + token + "\"}";
        given().contentType("application/json;charset=utf-8").body(body).when().post(base + "/contribute").then().statusCode(202);
    }

    @Test(timeout = 60000)
    public void contribute_withDeclinedToken_returnsBadRequest_3() {
        given().when().get(base + "/v2").then().statusCode(lessThan(300));
        String token = "tok_chargeDeclined_" + UUID.randomUUID().toString();
        String body = "{\"amount\":500,\"token\":\"" + token + "\"}";
        given().contentType("application/json;charset=utf-8").body(body).when().post(base + "/contribute").then().statusCode(400);
    }
}
