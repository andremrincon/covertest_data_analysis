package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CORSFilterTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void doFilterAddsAccessControlAllowOriginHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAddsAccessControlAllowMethodsHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Methods", "GET");
    }

    @Test(timeout = 60000)
    public void doFilterAddsAccessControlAllowHeadersHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Headers", "Accept, X-Requested-With");
    }

    @Test(timeout = 60000)
    public void doFilterAddsCacheControlHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Cache-Control", "public, max-age=86400");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToV2AllEndpoint() {
        given()
            .when()
                .get("/v2/all")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToNotFoundErrorResponse() {
        given()
            .when()
                .get("/v1/alpha/XYZ")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToBadRequestErrorResponse() {
        given()
            .when()
                .get("/v1/alpha/123")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterChainContinuesWithResponseBody() {
        given()
            .when()
                .get("/v1/alpha/US")
            .then()
                .body("alpha2Code", equalTo("US"));
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToNameEndpoint() {
        given()
            .when()
                .get("/v1/name/France")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToRegionEndpoint() {
        given()
            .when()
                .get("/v1/region/Europe")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToCapitalEndpoint() {
        given()
            .when()
                .get("/v1/capital/London")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doFilterAppliedToCurrencyEndpoint() {
        given()
            .when()
                .get("/v1/currency/USD")
            .then()
                .header("Access-Control-Allow-Origin", "*");
    }
}
