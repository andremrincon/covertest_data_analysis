package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class GammqTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
    }

    @Test(timeout = 60000)
    public void testGcfExecutionPath() {
        Response response = given()
            .when()
            .get("/api/gammq/5.5/1000.0");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserExecutionPath() {
        Response response = given()
            .when()
            .get("/api/gammq/5.5/2.3");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGserWithZeroX() {
        Response response = given()
            .when()
            .get("/api/gammq/5.5/0.0");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGammqWithNegativeA() {
        Response response = given()
            .when()
            .get("/api/gammq/-1.0/2.3");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGammqWithNegativeX() {
        Response response = given()
            .when()
            .get("/api/gammq/5.5/-1.0");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGserLargeAValue() {
        Response response = given()
            .when()
            .get("/api/gammq/1000.0/999.0");
        response.then().statusCode(lessThan(300));
    }
}
