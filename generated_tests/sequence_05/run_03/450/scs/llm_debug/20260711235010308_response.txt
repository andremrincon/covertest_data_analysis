package ts01qwen3_7_plus;

import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {

    @Test(timeout = 60000)
    public void testCostfuns_i5_sTest() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", 5, "test")
            .then()
                .statusCode(200)
                .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iMinus4_sAbab() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", -4, "abab")
            .then()
                .statusCode(200)
                .body(equalTo("0"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_iMinus500_sAbabba() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", -500, "ababba")
            .then()
                .statusCode(200)
                .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_i700_sZ() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", 700, "z")
            .then()
                .statusCode(200)
                .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_i0_sBaab() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", 0, "baab")
            .then()
                .statusCode(200)
                .body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void testCostfuns_invalidI() {
        given()
            .when()
                .get("http://localhost:8080/api/costfuns/{i}/{s}", "abc", "test")
            .then()
                .statusCode(400);
    }
}
