package ts01glm_5_2;

import io.restassured.RestAssured;

    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;

    @Test(timeout = 60000)
    public void testAddFeatureToProduct() {
        String productName = "Prod-" + UUID.randomUUID();
        String featureName = "Feat-" + UUID.randomUUID();
        RestAssured.given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        RestAssured.given().when().post("/products/" + productName + "/features/" + featureName).then().statusCode(201);
    }
