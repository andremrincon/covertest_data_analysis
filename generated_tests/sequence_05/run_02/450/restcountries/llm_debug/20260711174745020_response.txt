package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryRestV1Test {

    @Before
    public void setUp() {
        RestAssured.baseURI = System.getProperty("baseUrl", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void getByAlpha_returns200ForValidCode() {
        given().when().get("/v1/alpha/US").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByAlpha_returns400ForTooShortCode() {
        given().when().get("/v1/alpha/a").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByAlpha_returns404ForUnknownCode() {
        given().when().get("/v1/alpha/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByAlphaList_returns200ForValidCodes() {
        given().queryParam("codes", "US;CA").when().get("/v1/alpha/").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void getByAlphaList_returns400ForMissingCodesParam() {
        given().when().get("/v1/alpha/").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByAlphaList_returns400ForTooShortCodes() {
        given().queryParam("codes", "a").when().get("/v1/alpha/").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByAlphaList_returns404ForUnknownCodes() {
        given().queryParam("codes", "XX;YY;ZZ").when().get("/v1/alpha/").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByCurrency_returns400ForInvalidLength() {
        given().when().get("/v1/currency/US").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void getByCurrency_returns404ForUnknownCurrency() {
        given().when().get("/v1/currency/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByName_returns404ForUnknownName() {
        given().when().get("/v1/name/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByCallingCode_returns404ForUnknownCallingCode() {
        given().when().get("/v1/callingcode/99999").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByCapital_returns404ForUnknownCapital() {
        given().when().get("/v1/capital/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByRegion_returns404ForUnknownRegion() {
        given().when().get("/v1/region/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getBySubregion_returns404ForUnknownSubregion() {
        given().when().get("/v1/subregion/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void getByLanguage_returns404ForUnknownLanguage() {
        given().when().get("/v1/lang/123").then().statusCode(404);
    }
}
