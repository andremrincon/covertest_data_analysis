package ts01gpt_5_mini;

import io.restassured.response.Response;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class RegexTest {

    private static String base() {
        String b = System.getProperty("base.url");
        if (b == null || b.isEmpty()) {
            b = System.getenv("BASE_URL");
        }
        if (b == null || b.isEmpty()) {
            b = "http://localhost:8080";
        }
        return b;
    }

    @Test(timeout = 60000)
    public void testUrlPatternReturns200() {
        String base = base();
        given().when().get(base + "/api/pat/{txt}", "healthcheck").then().statusCode(lessThan(300));
        String txt = "http://a/b";
        Response resp = given().when().get(base + "/api/pat/{txt}", txt);
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDatePatternReturns200() {
        String base = base();
        given().when().get(base + "/api/pat/{txt}", "ping").then().statusCode(lessThan(300));
        String txt = "mon01jan";
        Response resp = given().when().get(base + "/api/pat/{txt}", txt);
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testFloatingPointExponentialPatternReturns200() {
        String base = base();
        given().when().get(base + "/api/pat/{txt}", "ready").then().statusCode(lessThan(300));
        String txt = "1.2e+03";
        Response resp = given().when().get(base + "/api/pat/{txt}", txt);
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testNonePatternReturns200() {
        String base = base();
        given().when().get(base + "/api/pat/{txt}", "start").then().statusCode(lessThan(300));
        String txt = "xyz";
        Response resp = given().when().get(base + "/api/pat/{txt}", txt);
        assertEquals(200, resp.getStatusCode());
    }
}
