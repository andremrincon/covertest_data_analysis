package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.notNullValue;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
        RestAssured.enableLoggingOfRequestAndResponseIfValidationFails();
    }

    @Test(timeout = 60000)
    public void v1AlphaByCodeReturnsCurrencyWithCodeNameSymbol() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/alpha/US")
        .then()
            .statusCode(200)
            .body("currencies[0]['code']", equalTo("USD"))
            .body("currencies[0]['name']", notNullValue())
            .body("currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v1CurrencyByCodeReturnsCountriesWithCurrencyFields() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/currency/USD")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", equalTo("USD"))
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v1NameReturnsCurrencyFieldsPopulated() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/name/France")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", equalTo("EUR"))
            .body("[0].currencies[0]['name']", equalTo("Euro"))
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v1AllReturnsCountriesWithCurrencyObjects() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/all")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", notNullValue())
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v2AlphaByCodeReturnsCurrencyWithAllFields() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/alpha/US")
        .then()
            .statusCode(200)
            .body("currencies[0]['code']", equalTo("USD"))
            .body("currencies[0]['name']", notNullValue())
            .body("currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v2CurrencyByCodeReturnsCountriesWithCurrencyFields() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v2/currency/EUR")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", equalTo("EUR"))
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v2RegionalblocWithCurrenciesFieldReturnsCurrencyObjects() {
        given()
            .accept(ContentType.JSON)
            .queryParam("fields", "name;capital;currencies")
        .when()
            .get("/v2/regionalbloc/EU")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", notNullValue())
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v1CallingcodeReturnsCountriesWithCurrencyFields() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/callingcode/1")
        .then()
            .statusCode(200)
            .body("[0].currencies[0]['code']", notNullValue())
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }

    @Test(timeout = 60000)
    public void v1CapitalReturnsCountriesWithCurrencyFields() {
        given()
            .accept(ContentType.JSON)
        .when()
            .get("/v1/capital/London")
        .then()

            .body("[0].currencies[0]['code']", equalTo("GBP"))
            .body("[0].currencies[0]['name']", notNullValue())
            .body("[0].currencies[0]['symbol']", notNullValue());
    }
}
