package ts01gpt_5_mini;

import org.junit.BeforeClass;
import org.junit.Test;
import io.restassured.response.Response;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.hamcrest.Matchers.equalTo;

public class CostfunsTest {
    private static String BASE;

    @BeforeClass
    public static void setup() {
        String env = System.getenv("API_BASE");
        if (env != null && !env.isEmpty()) {
            BASE = env;
        } else {
            BASE = System.getProperty("api.base", "http://localhost:8080");
        }
    }

    @Test(timeout = 60000)
    public void test_iEquals5_with_abab_returns_6() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "1", "1").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", 5, "abab");
        act.then().body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void test_iLessThanMinus444_returns_10() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "15.5", "4.5").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", -500, "a");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_iEqualsMinus333_with_abab_returns_6() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "subtract", "4.5", "0").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", -333, "abab");
        act.then().body(equalTo("6"));
    }

    @Test(timeout = 60000)
    public void test_iGreaterThan666_with_foo_returns_10() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "0", "0").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", 667, "foo");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_iEquals555_with_baab_returns_10() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "add", "15.5", "4.5").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", 555, "baab");
        act.then().body(equalTo("10"));
    }

    @Test(timeout = 60000)
    public void test_iEqualsMinus4_with_abab_returns_0() {
        given().baseUri(BASE).when().get("/api/calc/{op}/{arg1}/{arg2}", "subtract", "1", "1").then().statusCode(lessThan(300));
        Response act = given().baseUri(BASE).when().get("/api/costfuns/{i}/{s}", -4, "abab");
        act.then().body(equalTo("0"));
    }
}
