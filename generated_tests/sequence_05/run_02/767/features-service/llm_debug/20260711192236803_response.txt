    @Test(timeout = 60000)
    public void testDeleteConstraintById() {
        String productName = "test-prod-delete-constraint-id-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "src-feat-" + UUID.randomUUID().toString().substring(0, 8);
        String excludedFeature = "exc-feat-" + UUID.randomUUID().toString().substring(0, 8);

        Response r1 = post("/products/" + productName, null);
        assertTrue(r1.getStatusCode() < 300);

        Response r2 = post("/products/" + productName + "/features/" + sourceFeature, null);
        assertTrue(r2.getStatusCode() < 300);

        Response r3 = post("/products/" + productName + "/features/" + excludedFeature, null);
        assertTrue(r3.getStatusCode() < 300);

        Map<String, String> params = new HashMap<>();
        params.put("sourceFeature", sourceFeature);
        params.put("excludedFeature", excludedFeature);
        Response constraintResponse = post("/products/" + productName + "/constraints/excludes", params);

        assertTrue(constraintResponse.getStatusCode() < 300);

        String constraintId = extractJsonField(constraintResponse.getBody(), "id");
        if (constraintId == null) {
            constraintId = extractJsonField(constraintResponse.getBody(), "constraintId");
        }

        if (constraintId == null) {
            throw new AssertionError("Constraint id not found in response: " + constraintResponse.getBody());
        }

        Response del = delete("/products/" + productName + "/constraints/" + constraintId);
        assertEquals(204, del.getStatusCode());
    }

    private static Response post(String path, Map<String, String> formParams) {
        try {
            URL url = new URL(BASE_URL + path);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("POST");
            if (formParams != null && !formParams.isEmpty()) {
                conn.setDoOutput(true);
                conn.setRequestProperty("Content-Type", "application/x-www-form-urlencoded; charset=UTF-8");
                String form = buildForm(formParams);
                byte[] out = form.getBytes("UTF-8");
                conn.setRequestProperty("Content-Length", String.valueOf(out.length));
                OutputStream os = conn.getOutputStream();
                os.write(out);
                os.flush();
                os.close();
            } else {
                conn.setDoOutput(false);
            }
            int status = conn.getResponseCode();
            String body = readBody(conn);
            conn.disconnect();
            return new Response(status, body);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private static Response delete(String path) {
        try {
            URL url = new URL(BASE_URL + path);
            HttpURLConnection conn = (HttpURLConnection) url.openConnection();
            conn.setRequestMethod("DELETE");
            conn.setDoOutput(false);
            int status = conn.getResponseCode();
            String body = readBody(conn);
            conn.disconnect();
            return new Response(status, body);
        } catch (Exception e) {
            throw new RuntimeException(e);
        }
    }

    private static String buildForm(Map<String, String> params) throws Exception {
        StringBuilder sb = new StringBuilder();
        boolean first = true;
        for (Map.Entry<String, String> e : params.entrySet()) {
            if (!first) {
                sb.append("&");
            }
            first = false;
            sb.append(URLEncoder.encode(e.getKey(), "UTF-8"));
            sb.append("=");
            sb.append(URLEncoder.encode(e.getValue(), "UTF-8"));
        }
        return sb.toString();
    }

    private static String readBody(HttpURLConnection conn) {
        try {
            InputStream is;
            try {
                is = conn.getInputStream();
            } catch (Exception ex) {
                is = conn.getErrorStream();
            }
            if (is == null) {
                return "";
            }
            BufferedReader in = new BufferedReader(new InputStreamReader(is, "UTF-8"));
            String inputLine;
            StringBuilder content = new StringBuilder();
            while ((inputLine = in.readLine()) != null) {
                content.append(inputLine);
            }
            in.close();
            return content.toString();
        } catch (Exception e) {
            return "";
        }
    }

    private static String extractJsonField(String json, String field) {
        if (json == null) {
            return null;
        }
        String quotedField = "\"" + field + "\"";
        int idx = json.indexOf(quotedField);
        if (idx == -1) {
            return null;
        }
        int colon = json.indexOf(":", idx + quotedField.length());
        if (colon == -1) {
            return null;
        }
        int start = colon + 1;
        while (start < json.length() && Character.isWhitespace(json.charAt(start))) {
            start++;
        }
        if (start >= json.length()) {
            return null;
        }
        if (json.charAt(start) == '"') {
            start++;
            StringBuilder sb = new StringBuilder();
            for (int i = start; i < json.length(); i++) {
                char c = json.charAt(i);
                if (c == '"') {
                    return sb.toString();
                }
                if (c == '\\' && i + 1 < json.length()) {
                    i++;
                    sb.append(json.charAt(i));
                } else {
                    sb.append(c);
                }
            }
            return null;
        } else {
            StringBuilder sb = new StringBuilder();
            for (int i = start; i < json.length(); i++) {
                char c = json.charAt(i);
                if (c == ',' || c == '}' || Character.isWhitespace(c)) {
                    break;
                }
                sb.append(c);
            }
            return sb.toString();
        }
    }

    private static class Response {
        private final int statusCode;
        private final String body;

        Response(int statusCode, String body) {
            this.statusCode = statusCode;
            this.body = body == null ? "" : body;
        }

        int getStatusCode() {
            return statusCode;
        }

        String getBody() {
            return body;
        }
    }
