    private String formatPath(String path, Object... args) {
        Pattern p = Pattern.compile("\\{[^}]+\\}");
        Matcher m = p.matcher(path);
        StringBuffer sb = new StringBuffer();
        int i = 0;
        while (m.find()) {
            String replacement = "";
            if (i < args.length && args[i] != null) {
                try {
                    replacement = URLEncoder.encode(String.valueOf(args[i]), "UTF-8");
                } catch (Exception e) {
                    replacement = String.valueOf(args[i]);
                }
            }
            m.appendReplacement(sb, replacement);
            i++;
        }
        m.appendTail(sb);
        return sb.toString();
    }

    private Response send(String method, String path, Object... args) throws Exception {
        String formatted = formatPath(path, args);
        String urlString = baseUrl + formatted;
        URL url = new URL(urlString);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        conn.setConnectTimeout(30000);
        conn.setReadTimeout(30000);
        if ("POST".equals(method) || "PUT".equals(method)) {
            conn.setDoOutput(true);
            conn.setRequestProperty("Content-Length", "0");
        }
        int status = conn.getResponseCode();
        InputStream is;
        if (status >= 200 && status <= 299) {
            is = conn.getInputStream();
        } else {
            is = conn.getErrorStream();
            if (is == null) {
                is = conn.getInputStream();
            }
        }
        String body = "";
        if (is != null) {
            BufferedReader br = new BufferedReader(new InputStreamReader(is, "UTF-8"));
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) {
                sb.append(line).append("\n");
            }
            body = sb.toString().trim();
            br.close();
        }
        conn.disconnect();
        return new Response(status, body);
    }

    private Response post(String path, Object... args) throws Exception {
        return send("POST", path, args);
    }

    private Response get(String path, Object... args) throws Exception {
        return send("GET", path, args);
    }

    private Response delete(String path, Object... args) throws Exception {
        return send("DELETE", path, args);
    }

    private static class Response {
        private final int status;
        private final String body;

        Response(int status, String body) {
            this.status = status;
            this.body = body;
        }

        int statusCode() {
            return status;
        }

        String body() {
            return body;
        }
    }
