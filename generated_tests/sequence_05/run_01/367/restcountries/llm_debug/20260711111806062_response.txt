package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;
import static org.hamcrest.Matchers.lessThan;

public class CurrencyTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrencyCode() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies.USD.code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrencyName() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies.USD.name", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2AlphaByCodeReturnsCurrencySymbol() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/alpha/US")
                .then()
                .statusCode(200)
                .body("currencies.USD.symbol", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCountriesWithCurrencyCode() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/currency/EUR")
                .then()
                .statusCode(200)
                .body("currencies.EUR.code", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCountriesWithCurrencyName() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/currency/EUR")
                .then()
                .statusCode(200)
                .body("currencies.EUR.name", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV2CurrencyEndpointReturnsCountriesWithCurrencySymbol() {
        given()
                .queryParam("fields", "currencies")
                .when()
                .get("/v2/currency/EUR")
                .then()
                .statusCode(200)
                .body("currencies.EUR.symbol", notNullValue());
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsCurrencyCode() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body(containsString("USD"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsCurrencyName() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body(containsString("United States dollar"));
    }

    @Test(timeout = 60000)
    public void testV1AlphaByCodeReturnsCurrencySymbol() {
        given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200)
                .body(containsString("$"));
    }
}
