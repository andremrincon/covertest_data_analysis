package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class LanguageTest {

    @BeforeClass
    public static void setup() {
        String cfg = System.getProperty("rest.base.url");
        if (cfg == null || cfg.isEmpty()) {
            cfg = System.getenv("REST_BASE_URL");
        }
        if (cfg == null || cfg.isEmpty()) {
            cfg = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = cfg;
    }

    @Test(timeout = 60000)
    public void testSetIso639_1FromV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        eu.fayder.restcountries.v2.domain.Language lang = resp.jsonPath().getObject("languages[0]", eu.fayder.restcountries.v2.domain.Language.class);
        assertEquals("en", lang.getIso639_1());
    }

    @Test(timeout = 60000)
    public void testSetIso639_2FromV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        eu.fayder.restcountries.v2.domain.Language lang = resp.jsonPath().getObject("languages[0]", eu.fayder.restcountries.v2.domain.Language.class);
        assertEquals("eng", lang.getIso639_2());
    }

    @Test(timeout = 60000)
    public void testSetNameFromV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        eu.fayder.restcountries.v2.domain.Language lang = resp.jsonPath().getObject("languages[0]", eu.fayder.restcountries.v2.domain.Language.class);
        assertEquals("English", lang.getName());
    }

    @Test(timeout = 60000)
    public void testSetNativeNameFromV1Alpha_US() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        eu.fayder.restcountries.v2.domain.Language lang = resp.jsonPath().getObject("languages[0]", eu.fayder.restcountries.v2.domain.Language.class);
        assertEquals("English", lang.getNativeName());
    }
}
