package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import java.util.Arrays;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class WrongProductConfigurationExceptionTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("baseUrl");
        if (base == null || base.isEmpty()) base = System.getenv("BASE_URL");
        if (base == null || base.isEmpty()) base = "http://localhost:8080";
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testAddConfigurationWithVeryLongNameTriggersServerError() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        char[] arr = new char[800];
        Arrays.fill(arr, 'a');
        String longConfig = new String(arr);
        Response resp = given().when().post("/products/{productName}/configurations/{configurationName}", product, longConfig).then().extract().response();
        assertEquals(500, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddFeatureWithExcessivelyLongNameReturnsServerError() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        char[] arr = new char[1000];
        Arrays.fill(arr, 'x');
        String longFeature = new String(arr);
        Response resp = given().when().post("/products/{productName}/features/{featureName}", product, longFeature).then().extract().response();
        assertEquals(500, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfigurationWithJsonLikeFeatureNameReturnsServerError() {
        String product = "prod-" + UUID.randomUUID().toString();
        String config = "cfg-" + UUID.randomUUID().toString();
        given().when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/configurations/{configurationName}", product, config).then().statusCode(lessThan(300));
        String jsonLike = "{\"feature\":\"new-feature\"}";
        Response resp = given().when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, jsonLike).then().extract().response();
        assertEquals(500, resp.getStatusCode());
    }
}
