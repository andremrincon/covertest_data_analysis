package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.builder.RequestSpecBuilder;
import io.restassured.specification.RequestSpecification;
import org.junit.BeforeClass;
import org.junit.Test;

import java.util.UUID;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ConstraintExcludesTest {

    private static RequestSpecification spec;

    @BeforeClass
    public static void setup() {
        String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";
        RestAssured.baseURI = baseUrl;
        spec = new RequestSpecBuilder().build();
    }

    private String createProduct() {
        String productName = "prod-" + UUID.randomUUID().toString();
        given().spec(spec).when().post("/products/" + productName).then().statusCode(lessThan(300));
        return productName;
    }

    private void addFeature(String productName, String featureName) {
        given().spec(spec).when().post("/products/" + productName + "/features/" + featureName).then().statusCode(lessThan(300));
    }

    private void addConstraint(String productName, String source, String excluded) {
        given().spec(spec)
                .formParam("sourceFeature", source)
                .formParam("excludedFeature", excluded)
                .when()
                .post("/products/" + productName + "/constraints/excludes")
                .then().statusCode(lessThan(300));
    }

    private void createConfig(String productName, String configName) {
        given().spec(spec).when().post("/products/" + productName + "/configurations/" + configName).then().statusCode(lessThan(300));
    }

    private void addFeatureToConfig(String productName, String configName, String featureName) {
        given().spec(spec).when().post("/products/" + productName + "/configurations/" + configName + "/features/" + featureName).then().statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void testCreateExcludesConstraint() {
        String productName = createProduct();
        String sourceFeature = "source-" + UUID.randomUUID().toString();
        String excludedFeature = "excluded-" + UUID.randomUUID().toString();
        addFeature(productName, sourceFeature);
        addFeature(productName, excludedFeature);

        given().spec(spec)
                .formParam("sourceFeature", sourceFeature)
                .formParam("excludedFeature", excludedFeature)
                .when()
                .post("/products/" + productName + "/constraints/excludes")
                .then()
                .statusCode(201);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigBothFeaturesActive() {
        String productName = createProduct();
        String sourceFeature = "source-" + UUID.randomUUID().toString();
        String excludedFeature = "excluded-" + UUID.randomUUID().toString();
        addFeature(productName, sourceFeature);
        addFeature(productName, excludedFeature);
        addConstraint(productName, sourceFeature, excludedFeature);
        String configName = "config-" + UUID.randomUUID().toString();
        createConfig(productName, configName);
        addFeatureToConfig(productName, configName, sourceFeature);
        addFeatureToConfig(productName, configName, excludedFeature);

        given().spec(spec)
                .when()
                .get("/products/" + productName + "/configurations/" + configName)
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigOnlySourceActive() {
        String productName = createProduct();
        String sourceFeature = "source-" + UUID.randomUUID().toString();
        String excludedFeature = "excluded-" + UUID.randomUUID().toString();
        addFeature(productName, sourceFeature);
        addFeature(productName, excludedFeature);
        addConstraint(productName, sourceFeature, excludedFeature);
        String configName = "config-" + UUID.randomUUID().toString();
        createConfig(productName, configName);
        addFeatureToConfig(productName, configName, sourceFeature);

        given().spec(spec)
                .when()
                .get("/products/" + productName + "/configurations/" + configName)
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigOnlyExcludedActive() {
        String productName = createProduct();
        String sourceFeature = "source-" + UUID.randomUUID().toString();
        String excludedFeature = "excluded-" + UUID.randomUUID().toString();
        addFeature(productName, sourceFeature);
        addFeature(productName, excludedFeature);
        addConstraint(productName, sourceFeature, excludedFeature);
        String configName = "config-" + UUID.randomUUID().toString();
        createConfig(productName, configName);
        addFeatureToConfig(productName, configName, excludedFeature);

        given().spec(spec)
                .when()
                .get("/products/" + productName + "/configurations/" + configName)
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testEvaluateConfigNeitherActive() {
        String productName = createProduct();
        String sourceFeature = "source-" + UUID.randomUUID().toString();
        String excludedFeature = "excluded-" + UUID.randomUUID().toString();
        addFeature(productName, sourceFeature);
        addFeature(productName, excludedFeature);
        addConstraint(productName, sourceFeature, excludedFeature);
        String configName = "config-" + UUID.randomUUID().toString();
        createConfig(productName, configName);

        given().spec(spec)
                .when()
                .get("/products/" + productName + "/configurations/" + configName)
                .then()
                .statusCode(200);
    }
}
