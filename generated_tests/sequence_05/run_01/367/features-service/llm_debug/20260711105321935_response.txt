    private static class HttpResponse {
        int status;
        String body;

        HttpResponse(int status, String body) {
            this.status = status;
            this.body = body;
        }
    }

    private HttpResponse sendRequest(String method, String path, String contentType, String body) throws Exception {
        URL url = new URL(baseUrl + path);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod(method);
        if (contentType != null) {
            conn.setRequestProperty("Content-Type", contentType);
        }
        conn.setDoInput(true);
        if (body != null && (method.equals("POST") || method.equals("PUT"))) {
            conn.setDoOutput(true);
            byte[] data = body.getBytes("UTF-8");
            conn.setRequestProperty("Content-Length", Integer.toString(data.length));
            OutputStream os = conn.getOutputStream();
            os.write(data);
            os.flush();
            os.close();
        }
        int status = conn.getResponseCode();
        InputStream is;
        try {
            is = (status >= 400) ? conn.getErrorStream() : conn.getInputStream();
        } catch (Exception e) {
            is = null;
        }
        String responseBody = "";
        if (is != null) {
            BufferedReader br = new BufferedReader(new InputStreamReader(is, "UTF-8"));
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = br.readLine()) != null) {
                sb.append(line);
            }
            br.close();
            responseBody = sb.toString();
        }
        conn.disconnect();
        return new HttpResponse(status, responseBody);
    }

    private HttpResponse postNoBody(String path) throws Exception {
        return sendRequest("POST", path, null, null);
    }

    private HttpResponse postForm(String path, String[][] params) throws Exception {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < params.length; i++) {
            if (i > 0) sb.append("&");
            sb.append(URLEncoder.encode(params[i][0], "UTF-8"));
            sb.append("=");
            sb.append(URLEncoder.encode(params[i][1], "UTF-8"));
        }
        return sendRequest("POST", path, "application/x-www-form-urlencoded", sb.toString());
    }

    private HttpResponse get(String path) throws Exception {
        return sendRequest("GET", path, null, null);
    }

    private HttpResponse delete(String path) throws Exception {
        return sendRequest("DELETE", path, null, null);
    }

    private void assertStatusLessThan(HttpResponse resp, int threshold) {
        assertTrue("Expected status < " + threshold + " but was " + resp.status, resp.status < threshold);
    }

    private String extractJsonFieldString(String json, String key) {
        if (json == null) return null;
        String pattern = "\""+Pattern.quote(key)+"\"\\s*:\\s*\"([^\"]+)\"";
        Pattern p = Pattern.compile(pattern);
        Matcher m = p.matcher(json);
        if (m.find()) {
            return m.group(1);
        }
        pattern = "\""+Pattern.quote(key)+"\"\\s*:\\s*([^,}\\]]+)";
        p = Pattern.compile(pattern);
        m = p.matcher(json);
        if (m.find()) {
            return m.group(1).trim().replaceAll("^\"|\"$", "");
        }
        return null;
    }

    private Boolean extractJsonFieldBoolean(String json, String key) {
        String val = extractJsonFieldString(json, key);
        if (val == null) return null;
        if ("true".equalsIgnoreCase(val)) return Boolean.TRUE;
        if ("false".equalsIgnoreCase(val)) return Boolean.FALSE;
        return null;
    }

    }
