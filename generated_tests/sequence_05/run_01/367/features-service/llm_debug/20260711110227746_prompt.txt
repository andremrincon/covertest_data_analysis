        spec = new RequestSpecBuilder().build();
    }

    private String createProduct() {
        String productName = "prod-" + UUID.randomUUID().toString();
        given().spec(spec).when().post("/products/" + productName).then().statusCode(lessThan(300));

        given().spec(spec).when().post("/products/" + productName + "/features/" + featureName).then().statusCode(lessThan(300));
    }

        given().spec(spec)

                .then().statusCode(lessThan(300));
    }
