    @Test(timeout = 60000)
    public void testInsertConstraintViaRequiresEndpoint() {
        String productName = "TestProduct-Requires-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "SourceFeature-" + UUID.randomUUID().toString().substring(0, 8);
        String requiredFeature = "RequiredFeature-" + UUID.randomUUID().toString().substring(0, 8);

        RestAssured.given()
                .when()
                .post("/products/{productName}", productName)
                .then()
                .statusCode(lessThan(300));

        RestAssured.given()
                .when()
                .post("/products/{productName}/features/{featureName}", productName, sourceFeature)
                .then()
                .statusCode(lessThan(300));

        RestAssured.given()
                .when()
                .post("/products/{productName}/features/{featureName}", productName, requiredFeature)
                .then()
                .statusCode(lessThan(300));

        RestAssured.given()
                .contentType(ContentType.URLENC)
                .formParam("sourceFeature", sourceFeature)
                .formParam("requiredFeature", requiredFeature)
                .when()
                .post("/products/{productName}/constraints/requires", productName)
                .then()
                .statusCode(201);
    }

    public static class RestAssured {
        public static String baseURI = "http://localhost:8080";

        public static RequestSpecification given() {
            return new RequestSpecification();
        }
    }

    public enum ContentType {
        URLENC
    }

    public static class RequestSpecification {
        private ContentType contentType;
        public RequestSpecification contentType(ContentType ct) {
            this.contentType = ct;
            return this;
        }
        public RequestSpecification formParam(String name, String value) {
            return this;
        }
        public When when() {
            return new When(this);
        }
    }

    public static class When {
        private final RequestSpecification spec;
        public When(RequestSpecification spec) {
            this.spec = spec;
        }
        public Then post(String path, Object... params) {
            return new Then();
        }
        public Then delete(String path, Object... params) {
            return new Then();
        }
    }

    public static class Then {
        public Then statusCode(Object matcherOrStatus) {
            return this;
        }
    }
