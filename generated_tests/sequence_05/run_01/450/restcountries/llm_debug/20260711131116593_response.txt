package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class CountryRestV1Test {

    private static final String BASE_URL = System.getProperty("base.url", "http://localhost:8080/rest");

    @BeforeClass
    public static void setup() {
        RestAssured.baseURI = BASE_URL;
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_ValidAlphaCode_Returns200() {
        Response response = given()
                .when()
                .get("/v1/alpha/US");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_InvalidAlphaCode_Returns400() {
        Response response = given()
                .when()
                .get("/v1/alpha/1");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_ValidCodes_Returns200() {
        Response response = given()
                .queryParam("codes", "US;CA")
                .when()
                .get("/v1/alpha/");
        response.then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_EmptyCodes_Returns400() {
        Response response = given()
                .queryParam("codes", "")
                .when()
                .get("/v1/alpha/");
        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_ValidCurrency_Returns200() {
