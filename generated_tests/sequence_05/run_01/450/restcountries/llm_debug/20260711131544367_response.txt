package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class StripeRestTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("BASE_URL", System.getenv("BASE_URL"));
        if (base == null || base.trim().isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testContributeWithEmptyTokenReturnsBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String body = "{\"amount\":100,\"token\":\"\"}";
        Response res = given().contentType("application/json;charset=utf-8").body(body).when().post("/contribute");
        assertEquals(400, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testContributeWithNoBodyReturnsBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response res = given().contentType("application/json;charset=utf-8").when().post("/contribute");
        assertEquals(400, res.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testContributeWithValidTokenReturnsAccepted() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        String body = "{\"amount\":150,\"token\":\"" + token + "\"}";
        Response res = given().contentType("application/json;charset=utf-8").body(body).when().post("/contribute");
        assertEquals(202, res.getStatusCode());
    }
}
