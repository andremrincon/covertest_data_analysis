package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class DateParseTest {

    @BeforeClass
    public static void setUp() {
        String base = System.getProperty("api.base", System.getenv().getOrDefault("API_BASE", "http://localhost:8080"));
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void test_dateparse_success_wednesday_august_returns_200() {
        given().when().get("/api/pat/{txt}", "The quick brown fox").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "Wednesday", "August");
        assertEquals(200, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void test_dateparse_mon_jan_returns_combined_value_2() {
        given().when().get("/api/pat/{txt}", "The quick brown fox").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "mon", "jan");
        assertEquals("2", resp.asString());
    }

    @Test(timeout = 60000)
    public void test_dateparse_invalid_day_dec_returns_12() {
        given().when().get("/api/pat/{txt}", "Sample text for arrange").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "notaday", "dec");
        assertEquals("12", resp.asString());
    }

    @Test(timeout = 60000)
    public void test_dateparse_tue_unknown_month_returns_1() {
        given().when().get("/api/pat/{txt}", "Arrange call").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "tue", "unknownmonth");
        assertEquals("1", resp.asString());
    }

    @Test(timeout = 60000)
    public void test_dateparse_numeric_inputs_return_500() {
        given().when().get("/api/pat/{txt}", "Arrange before error case").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "123", "456");
        assertEquals(500, resp.getStatusCode());
    }

    @Test(timeout = 60000)
    public void test_dateparse_case_insensitive_month_mar_returns_200() {
        given().when().get("/api/pat/{txt}", "Setup for case test").then().statusCode(lessThan(300));
        Response resp = given().when().get("/api/dateparse/{dayname}/{monthname}", "tuesday", "MAR");
        assertEquals(200, resp.getStatusCode());
    }
}
