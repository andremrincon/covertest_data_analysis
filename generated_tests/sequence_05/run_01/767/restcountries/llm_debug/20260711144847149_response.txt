package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class StripeRestTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("rest.base");
        if (base == null || base.isEmpty()) {
            base = System.getenv("REST_BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testContributeMissingTokenReturnsBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String payload = "{\"amount\":100}";
        Response act = given()
                .contentType("application/json;charset=utf-8")
                .body(payload)
                .when()
                .post("/contribute");
        act.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeWithTokenTriggersStripeErrorReturnsBadRequest() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String token = "tok_" + UUID.randomUUID().toString();
        String payload = "{\"amount\":150,\"token\":\"" + token + "\"}";
        Response act = given()
                .contentType("application/json;charset=utf-8")
                .body(payload)
                .when()
                .post("/contribute");
        act.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeAcceptedWhenChargeSucceeds() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        String token = "success_" + UUID.randomUUID().toString();
        String payload = "{\"amount\":200,\"token\":\"" + token + "\"}";
        Response act = given()
                .contentType("application/json;charset=utf-8")
                .body(payload)
                .when()
                .post("/contribute");
        assertEquals(202, act.getStatusCode());
    }
}
