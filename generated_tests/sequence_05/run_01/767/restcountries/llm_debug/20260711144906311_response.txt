package ts01gpt_5_mini;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class CountryRestV1Test {

    @BeforeClass
    public static void setup() {
        String env = System.getenv("API_BASE_URL");
        if (env == null || env.isEmpty()) {
            env = System.getProperty("api.base", "http://localhost:8080/rest");
        }
        RestAssured.baseURI = env;
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_Success_US_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/US").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_BadRequest_numeric_returns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlpha_NotFound_XYZ_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/alpha/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_Success_codes_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "US,CA,MX").when().get("/v1/alpha").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_BadRequest_short_codes_returns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "1").when().get("/v1/alpha").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_NotFound_codes_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "true,XX,YY,ZZ").when().get("/v1/alpha").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByAlphaList_InternalServerError_malformed_returns500() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("codes", "[\"US\",\"CA\"]").when().get("/v1/alpha").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_Success_USD_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/USD").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_BadRequest_numeric_returns400() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/123").then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_NotFound_XYZ_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/XYZ").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByCurrency_InternalServerError_example_returns500() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/currency/EUR").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testGetByName_Success_France_fullTextFalse_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().queryParam("fullText", "false").when().get("/v1/name/France").then().statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetByName_NotFound_numeric_returns404() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/name/123").then().statusCode(404);
    }

    @Test(timeout = 60000)
    public void testGetByName_InternalServerError_example_returns500() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/name/True").then().statusCode(500);
    }

    @Test(timeout = 60000)
    public void testGetByCallingCode_Success_1_returns200() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        given().when().get("/v1/callingcode/1").then().statusCode(200);
    }
}
