package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class ExpintTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = System.getenv("BASE_URL");
        }
        if (baseUrl == null || baseUrl.isEmpty()) {
            baseUrl = "http://localhost:8080";
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testExpint_n0_x0_throwsException() {
        given()
                .pathParam("n", 0)
                .pathParam("x", 0)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testExpint_n0_positiveX_returns200() {
        given()
                .pathParam("n", 0)
                .pathParam("x", 2.5)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpint_nGreaterThan1_xZero_returns200() {
        given()
                .pathParam("n", 3)
                .pathParam("x", 0)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpint_nGreaterThan1_xGreaterThanOne_returns200() {
        given()
                .pathParam("n", 3)
                .pathParam("x", 2.5)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpint_nGreaterThan1_xLessThanOne_nm1NotZero_returns200() {
        given()
                .pathParam("n", 3)
                .pathParam("x", 0.1)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testExpint_n1_xLessThanOne_nm1Zero_returns200() {
        given()
                .pathParam("n", 1)
                .pathParam("x", 0.1)
        .when()
                .get("/api/expint/{n}/{x}")
        .then()
                .statusCode(200);
    }
}
