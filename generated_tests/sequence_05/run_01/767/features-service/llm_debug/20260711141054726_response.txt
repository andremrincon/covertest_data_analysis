package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.Method;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class CORSFilterTest {

    private static String baseUrl;

    @BeforeClass
    public static void setUp() {
        baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void doGetRequestTriggersDoFilterChainDoFilter() {
        String productName = "CORS-Test-Product-" + java.util.UUID.randomUUID().toString().substring(0, 8);

        given()
                .when()
                .request(Method.OPTIONS, "/products/" + productName)
                .then()
                .statusCode(lessThan(500));

        given()
                .when()
                .get("/products")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void doOptionsRequestTriggersDoFilterWithoutChainDoFilter() {
        given()
                .when()
                .request(Method.OPTIONS, "/products")
                .then()
                .statusCode(lessThan(500));
    }

    @Test(timeout = 60000)
    public void doPostRequestTriggersDoFilterChainDoFilter() {
        String productName = "CORS-Post-Product-" + java.util.UUID.randomUUID().toString().substring(0, 8);

        given()
                .when()
                .post("/products/" + productName)
                .then()
                .statusCode(lessThan(300));
    }

    @Test(timeout = 60000)
    public void doDeleteRequestTriggersDoFilterChainDoFilter() {
        String productName = "CORS-Delete-Product-" + java.util.UUID.randomUUID().toString().substring(0, 8);

        given()
                .when()
                .post("/products/" + productName)
                .then()
                .statusCode(lessThan(300));

        given()
                .when()
                .delete("/products/" + productName)
                .then()
                .statusCode(204);
    }

    @Test(timeout = 60000)
    public void doGetResponseHasAccessControlAllowOriginHeader() {
        given()
                .when()
                .get("/products")
                .then()
                .header("Access-Control-Allow-Origin", "*");
    }

    @Test(timeout = 60000)
    public void doOptionsResponseHasAccessControlAllowMethodsHeader() {
        given()
                .when()
                .request(Method.OPTIONS, "/products")
                .then()
                .header("Access-Control-Allow-Methods", "POST, PUT, GET, OPTIONS, DELETE");
    }
}
