    @Test(timeout = 60000)
    public void testDeleteRequiresConstraint() {
        String productName = "TestProduct-DelReq-" + UUID.randomUUID().toString().substring(0, 8);
        String sourceFeature = "SrcFeat-" + UUID.randomUUID().toString().substring(0, 8);
        String requiredFeature = "ReqFeat-" + UUID.randomUUID().toString().substring(0, 8);

        given().when().post("/products/{productName}", productName).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, sourceFeature).then().statusCode(lessThan(300));
        given().when().post("/products/{productName}/features/{featureName}", productName, requiredFeature).then().statusCode(lessThan(300));

        Response createResponse = given()
            .contentType(ContentType.URLENC)
            .formParam("sourceFeature", sourceFeature)
            .formParam("requiredFeature", requiredFeature)
        .when()
            .post("/products/{productName}/constraints/requires", productName);
        createResponse.then().statusCode(lessThan(300));

        String constraintId = null;
        String body = createResponse.asString();
        if (body != null && !body.trim().isEmpty() && (body.trim().startsWith("{") || body.trim().startsWith("["))) {
            constraintId = createResponse.jsonPath().getString("id");
            if (constraintId == null) {
                constraintId = createResponse.jsonPath().getString("constraintId");
            }
        }
        if (constraintId == null) {
            String location = createResponse.getHeader("Location");
            if (location != null) {
                int idx = location.lastIndexOf('/');
                if (idx != -1 && idx + 1 < location.length()) {
                    constraintId = location.substring(idx + 1);
                } else {
                    constraintId = location;
                }
            }
        }
        if (constraintId == null) {
            constraintId = createResponse.getHeader("id");
        }
        if (constraintId == null) {
            throw new AssertionError("Could not determine constraint id from create response");
        }

        given()
        .when()
            .delete("/products/{productName}/constraints/{constraintId}", productName, constraintId)
        .then()
            .statusCode(204);
    }
