import org.junit.Test;

public class FileSuffixTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080");
        RestAssured.baseURI = baseUrl;

    @Test(timeout = 60000)
    public void testNoDotInFileReturnsZero() {
        given()
            .pathParam("directory", "text")
