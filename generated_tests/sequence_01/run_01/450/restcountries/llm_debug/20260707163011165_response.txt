package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.Before;
import org.junit.Test;
import static org.hamcrest.Matchers.lessThan;

public class ContributionTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("test.base.url", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testContributeValid() {

        Response response = RestAssured.given()
                .contentType("application/json")
                .body("{\"amount\": 100, \"token\": \"tok_test_123\"}")
                .when()
                .post("/contribute");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeInvalidEmptyBody() {

        Response response = RestAssured.given()
                .contentType("application/json")
                .body("{}")
                .when()
                .post("/contribute");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeInvalidMissingToken() {

        Response response = RestAssured.given()
                .contentType("application/json")
                .body("{\"amount\": 100}")
                .when()
                .post("/contribute");

        response.then().statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeInvalidMissingAmount() {

        Response response = RestAssured.given()
                .contentType("application/json")
                .body("{\"token\": \"tok_test_123\"}")
                .when()
                .post("/contribute");

        response.then().statusCode(400);
    }
}
