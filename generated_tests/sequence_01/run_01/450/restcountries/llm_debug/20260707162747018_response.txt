package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class CORSFilterTest {

    @Before
    public void setUp() {
        RestAssured.baseURI = "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testCorsAllowOriginHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Origin", equalTo("*"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowMethodsHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Methods", equalTo("GET"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowHeadersHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Access-Control-Allow-Headers", equalTo("Accept, X-Requested-With"));
    }

    @Test(timeout = 60000)
    public void testCorsCacheControlHeaderOnV1All() {
        given()
            .when()
                .get("/v1/all")
            .then()
                .header("Cache-Control", equalTo("public, max-age=86400"));
    }
