package ts01qwen3_7_plus;

import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class DuplicatedObjectExceptionTest {

    @Test(timeout = 60000)
    public void testDuplicateProduct() {
        String productName = "DupProd-" + UUID.randomUUID().toString();

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName)
            .then()
            .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testDuplicateFeature() {
        String productName = "DupFeatProd-" + UUID.randomUUID().toString();
        String featureName = "DupFeat-" + UUID.randomUUID().toString();

        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/features/" + featureName).then().statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/features/" + featureName)
            .then()
            .statusCode(500);
    }

    @Test(timeout = 60000)
    public void testDuplicateConfiguration() {
        String productName = "DupConfProd-" + UUID.randomUUID().toString();
        String configName = "DupConf-" + UUID.randomUUID().toString();

        given().when().post("/products/" + productName).then().statusCode(lessThan(300));
        given().when().post("/products/" + productName + "/configurations/" + configName).then().statusCode(lessThan(300));

        given()
            .when()
            .post("/products/" + productName + "/configurations/" + configName)
            .then()
            .statusCode(500);
    }
}
