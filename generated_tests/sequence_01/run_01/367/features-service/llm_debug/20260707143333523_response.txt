package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import java.net.URLEncoder;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class ProductsConfigurationFeaturesResourceTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("api.base", System.getenv("API_BASE_URL"));
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void deleteFeature_shouldReturnNoContent_whenFeatureExistsInConfiguration() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String productName = "test-product-" + uuid;
        String configurationName = "cfg-" + uuid;
        String featureName = "feat-" + uuid;
        String p = URLEncoder.encode(productName, "UTF-8");
        String c = URLEncoder.encode(configurationName, "UTF-8");
        String f = URLEncoder.encode(featureName, "UTF-8");
        given().when().post("/products/" + p).then().statusCode(lessThan(300));
        given().when().post("/products/" + p + "/configurations/" + c).then().statusCode(lessThan(300));
        given().when().post("/products/" + p + "/configurations/" + c + "/features/" + f).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/" + p + "/configurations/" + c + "/features/" + f);
        act.then().statusCode(204);
    }

    @Test(timeout = 60000)
    public void deleteFeature_shouldReturnServerError_whenFeatureNameIsInvalidOrCausesEvaluationFailure() throws Exception {
        String uuid = UUID.randomUUID().toString();
        String productName = "test-product-" + uuid;
        String configurationName = "cfg-" + uuid;
        String longFeature = new String(new char[1200]).replace("\0", "x");
        String p = URLEncoder.encode(productName, "UTF-8");
        String c = URLEncoder.encode(configurationName, "UTF-8");
        String f = URLEncoder.encode(longFeature, "UTF-8");
        given().when().post("/products/" + p).then().statusCode(lessThan(300));
        given().when().post("/products/" + p + "/configurations/" + c).then().statusCode(lessThan(300));
        Response act = given().when().delete("/products/" + p + "/configurations/" + c + "/features/" + f);
        act.then().statusCode(500);
    }
}
