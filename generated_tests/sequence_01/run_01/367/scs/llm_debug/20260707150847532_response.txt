package ts01qwen3_7_plus;

import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;

public class RegexTest {

    private static final String BASE_URL = System.getenv().getOrDefault("BASE_URL", "http://localhost:8080");

    @Test(timeout = 60000)
    public void testUrlMatch() {
        given()
            .pathParam("txt", "http://a/b")
            .when()
                .get(BASE_URL + "/api/pat/{txt}")
            .then()
                .statusCode(200)
                .body(equalTo("url"));
    }

    @Test(timeout = 60000)
    public void testDateMatch() {
        given()
            .pathParam("txt", "mon01jan")
            .when()
                .get(BASE_URL + "/api/pat/{txt}")
            .then()
                .statusCode(200)
                .body(equalTo("date"));
    }

    @Test(timeout = 60000)
    public void testFpeMatch() {
        given()
            .pathParam("txt", "1.2e+34")
            .when()
                .get(BASE_URL + "/api/pat/{txt}")
            .then()
                .statusCode(200)
                .body(equalTo("fpe"));
    }

    @Test(timeout = 60000)
    public void testNoneMatch() {
        given()
            .pathParam("txt", "hello")
            .when()
                .get(BASE_URL + "/api/pat/{txt}")
            .then()
                .statusCode(200)
                .body(equalTo("none"));
    }
}
