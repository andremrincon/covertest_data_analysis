package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class NcsRestTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl");
        if (baseUrl != null && !baseUrl.isEmpty()) {
            RestAssured.baseURI = baseUrl;
        } else {
            RestAssured.baseURI = "http://localhost:8080";
        }
    }

    @Test(timeout = 60000)
    public void testClassifyWithNonPositiveFirstSide_1() {
        given()
                .when()
                .get("/api/triangle/0/4/5")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(0));
    }

    @Test(timeout = 60000)
    public void testClassifyWithNonPositiveSecondSide_1() {
        given()
                .when()
                .get("/api/triangle/3/-1/5")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(0));
    }

    @Test(timeout = 60000)
    public void testClassifyEquilateralTriangle_1() {
        given()
                .when()
                .get("/api/triangle/5/5/5")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(3));
    }

    @Test(timeout = 60000)
    public void testClassifyTriangleInequalityViolation_1() {
        given()
                .when()
                .get("/api/triangle/1/2/4")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(0));
    }

    @Test(timeout = 60000)
    public void testClassifyIsoscelesTriangle_1() {
        given()
                .when()
                .get("/api/triangle/3/3/4")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(2));
    }

    @Test(timeout = 60000)
    public void testClassifyScaleneTriangle_1() {
        given()
                .when()
                .get("/api/triangle/3/4/5")
                .then()
                .statusCode(200)
                .body("resultAsInt", equalTo(1));
    }
}
