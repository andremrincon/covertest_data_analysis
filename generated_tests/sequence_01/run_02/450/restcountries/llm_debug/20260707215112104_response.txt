package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.equalTo;
import static org.hamcrest.Matchers.lessThan;

public class ResponseEntityTest {

    @BeforeClass
    public static void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGetStatusOnNotFoundResponse() {
        given()
                .when()
                .get("/v1/name/123")
                .then()
                .statusCode(404)
                .body("status", equalTo(404));
    }

    @Test(timeout = 60000)
    public void testGetMessageOnNotFoundResponse() {
        given()
                .when()
                .get("/v1/capital/123")
                .then()
                .statusCode(404)
                .body("message", equalTo("Not Found"));
    }

    @Test(timeout = 60000)
    public void testGetStatusOnServerErrorResponse() {
        given()
                .when()
                .get("/v1/region/True")
                .then()
                .statusCode(500)
                .body("status", equalTo(500));
    }

    @Test(timeout = 60000)
    public void testGetMessageOnServerErrorResponse() {
        given()
                .when()
                .get("/v1/name/True")
                .then()
                .statusCode(500)
                .body("message", equalTo("Internal Server Error"));
    }

    @Test(timeout = 60000)
    public void testGetStatusOnCallingCodeNotFound() {
        given()
                .when()
                .get("/v1/callingcode/99999")
                .then()
                .statusCode(404)
                .body("status", equalTo(404));
    }

    @Test(timeout = 60000)
    public void testGetMessageOnCallingCodeServerError() {
        given()
                .when()
                .get("/v1/callingcode/True")
                .then()
                .statusCode(500)
                .body("message", equalTo("Internal Server Error"));
    }
}
