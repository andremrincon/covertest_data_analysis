package ts01glm_5_2;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.*;
import static org.hamcrest.Matchers.*;

public class PatTest {

    @Before
    public void setUp() {
        String baseUri = System.getProperty("base.url", "http://localhost:8080");
        RestAssured.baseURI = baseUri;
    }

    @Test(timeout = 60000)
    public void testPatFoundReverseNotFound() {
        given()
            .when()
                .get("/api/pat/abcxxxxx/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatFoundReverseFoundImmediately() {
        given()
            .when()
                .get("/api/pat/abccba/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatFoundReverseFoundWithGap() {
        given()
            .when()
                .get("/api/pat/abcxxxcba/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testReverseFoundFirstPatFoundImmediately() {
        given()
            .when()
                .get("/api/pat/cbaabc/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testReverseFoundFirstPatFoundWithGap() {
        given()
            .when()
                .get("/api/pat/cbaxxxabc/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testReverseFoundPatNotFound() {
        given()
            .when()
                .get("/api/pat/cbaxxxxx/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testPatLenNotGreaterThanTwo() {
        given()
            .when()
                .get("/api/pat/ab/ab")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testNeitherFound() {
        given()
            .when()
                .get("/api/pat/xxxxxxx/abc")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFirstCharMatchButFullPatternNoMatch() {
        given()
            .when()
                .get("/api/pat/abxcbz/abc")
            .then()
                .statusCode(200);
    }
}
