package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;

public class StripeRestTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        RestAssured.baseURI = baseUrl != null ? baseUrl : "http://localhost:8080/rest";
    }

    @Test(timeout = 60000)
    public void testContributeNullBody() {
        given()
            .contentType(ContentType.JSON)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeNullToken() {
        String json = "{\"amount\": 100, \"token\": null}";
        given()
            .contentType(ContentType.JSON)
            .body(json)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeEmptyToken() {
        String json = "{\"amount\": 100, \"token\": \"\"}";
        given()
            .contentType(ContentType.JSON)
            .body(json)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeBlankToken() {
        String json = "{\"amount\": 100, \"token\": \"   \"}";
        given()
            .contentType(ContentType.JSON)
            .body(json)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testContributeValidToken() {
        String json = "{\"amount\": 100, \"token\": \"tok_123456789\"}";
        given()
            .contentType(ContentType.JSON)
            .body(json)
        .when()
            .post("/contribute")
        .then()
            .statusCode(400);
    }
}
