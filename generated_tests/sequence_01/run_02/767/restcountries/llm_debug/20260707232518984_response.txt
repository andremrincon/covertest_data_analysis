package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Before;
import org.junit.Test;

public class CountryTranslationsTest {

    @Before
    public void setUp() {
        String baseUrl = System.getenv("BASE_URL");
        if (baseUrl == null) {
            baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        }
        RestAssured.baseURI = baseUrl;
    }

    @Test(timeout = 60000)
    public void testGetAlphaCodeReturnsTranslations() {
        RestAssured.given()
                .when()
                .get("/v1/alpha/US")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetNameReturnsTranslations() {
        RestAssured.given()
                .when()
                .get("/v1/name/France")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetCurrencyReturnsTranslations() {
        RestAssured.given()
                .when()
                .get("/v1/currency/USD")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetCapitalReturnsTranslations() {
        RestAssured.given()
                .when()
                .get("/v1/capital/London")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testGetRegionReturnsTranslations() {
        RestAssured.given()
                .when()

                .then()
                .statusCode(200);
    }
}
