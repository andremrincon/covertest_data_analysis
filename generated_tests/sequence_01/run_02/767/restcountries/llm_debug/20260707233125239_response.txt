package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CurrencyTest {

    @BeforeClass
    public static void setup() {
        String base = System.getProperty("base.url");
        if (base == null || base.isEmpty()) {
            base = System.getenv("BASE_URL");
        }
        if (base == null || base.isEmpty()) {
            base = "http://localhost:8080/rest";
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testSetCodeViaV1Alpha_US_returnsCurrencyWithCode() {
        given().when().get("/v1/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/alpha/US");
        resp.then().statusCode(200);
        eu.fayder.restcountries.v2.domain.Currency currency = resp.getBody().jsonPath().getObject("currencies[0]", eu.fayder.restcountries.v2.domain.Currency.class);
        assertEquals("USD", currency.getCode());
    }

    @Test(timeout = 60000)
    public void testSetNameViaV2Currency_EUR_returnsCurrencyWithName() {
        given().when().get("/v2/all").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v2/currency/EUR");
        resp.then().statusCode(200);
        eu.fayder.restcountries.v2.domain.Currency currency = resp.getBody().jsonPath().getObject("0.currencies[0]", eu.fayder.restcountries.v2.domain.Currency.class);
        assertEquals("Euro", currency.getName());
    }

    @Test(timeout = 60000)
    public void testSetSymbolViaV1All_mapShapeCurrencyHasSymbol() {
        given().when().get("/v1/alpha/GB").then().statusCode(lessThan(300));
        Response resp = given().when().get("/v1/all");
        resp.then().statusCode(200);
        eu.fayder.restcountries.v2.domain.Currency currency = resp.getBody().jsonPath().getObject("value[0].currencies.EUR", eu.fayder.restcountries.v2.domain.Currency.class);
        assertEquals("€", currency.getSymbol());
    }
}
