package ts01glm_5_2;

import io.restassured.RestAssured;
import io.restassured.http.ContentType;
import org.junit.Before;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.*;

public class ResponseEntityTest {

    @Before
    public void setUp() {
        String baseUrl = System.getProperty("baseUrl", "http://localhost:8080/rest");
        RestAssured.baseURI = baseUrl;
        RestAssured.enableLoggingOfRequestAndResponseIfValidationFails();
    }

    @Test(timeout = 60000)
    public void postV1ReturnsMethodNotAllowedWithStatusAndMessage() {
        given()
                .contentType(ContentType.JSON)
                .body("{}")
        .when()
                .post("/v1")
        .then()
                .statusCode(405)
                .body("status", equalTo(405))
                .body("message", notNullValue());
    }

    @Test(timeout = 60000)
    public void postV2ReturnsMethodNotAllowedWithStatusAndMessage() {
        given()
                .contentType(ContentType.JSON)
                .body("{}")
        .when()
                .post("/v2")
        .then()
                .statusCode(405)
                .body("status", equalTo(405))
                .body("message", notNullValue());
    }

    @Test(timeout = 60000)
    public void postContributeWithInvalidPayloadReturnsBadRequestWithStatusAndMessage() {
        given()
                .contentType(ContentType.JSON)
                .body("{}")
        .when()
                .post("/contribute")
        .then()
                .statusCode(400)
                .body("message", notNullValue());
    }

    @Test(timeout = 60000)
    public void postContributeWithMissingBodyReturnsBadRequestWithMessage() {
        given()
                .contentType(ContentType.JSON)
        .when()
                .post("/contribute")
        .then()
                .statusCode(400)
                .body("message", notNullValue());
    }

    @Test(timeout = 60000)
    public void postContributeWithMalformedJsonReturnsBadRequestWithMessage() {
        given()
                .contentType(ContentType.JSON)
                .body("invalid json")
        .when()
                .post("/contribute")
        .then()
                .statusCode(anyOf(equalTo(400), equalTo(500)))
                .body("message", notNullValue());
    }

    @Test(timeout = 60000)
    public void postV1WithEmptyBodyReturnsMethodNotAllowedWithStatus() {
        given()
                .contentType(ContentType.JSON)
                .body("")
        .when()
                .post("/v1")
        .then()
                .statusCode(405)
                .body("status", equalTo(405));
    }
}
