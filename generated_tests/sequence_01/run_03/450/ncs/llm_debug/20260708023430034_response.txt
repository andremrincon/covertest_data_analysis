package ts01qwen3_7_plus;

import static io.restassured.RestAssured.given;
import org.junit.Test;

public class FisherTest {

    private static final String BASE_URL = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testFisherA1B1() {
        given()
            .when()
                .get(BASE_URL + "/api/fisher/1/1/0.75")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherA1BNot1() {
        given()
            .when()
                .get(BASE_URL + "/api/fisher/1/2/0.75")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherPGreaterThan1() {
        given()
            .when()
                .get(BASE_URL + "/api/fisher/10/5/5000.0")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherPBetween0And1() {
        given()
            .when()
                .get(BASE_URL + "/api/fisher/10/5/0.75")
            .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testFisherXZero() {
        given()
            .when()
                .get(BASE_URL + "/api/fisher/1/1/0.0")
            .then()
                .statusCode(200);
    }
}
