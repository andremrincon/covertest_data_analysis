package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CalcTest {
    @BeforeClass
    public static void setupClass() {
        String base = System.getenv("TEST_BASE_URL");
        if (base == null || base.isEmpty()) {
            base = System.getProperty("api.base", "http://localhost:8080");
        }
        RestAssured.baseURI = base;
    }

    @Test(timeout = 60000)
    public void testPiReturnsConstantValue() {
        given().when().get("/api/pat/The").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/calc/pi/0/0");
        assertEquals(200, act.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testPlusOperationReturnsSumString() {
        given().when().get("/api/text2txt/The/quick/brown").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/calc/plus/15.5/4.5");
        assertEquals("20.0", act.getBody().asString());
    }

    @Test(timeout = 60000)
    public void testDivideByZeroProducesServerError() {
        given().when().get("/api/cookie/session-id/1/example.com").then().statusCode(lessThan(300));
        Response act = given().when().get("/api/calc/divide/100/0");
        assertEquals(500, act.getStatusCode());
    }
}
