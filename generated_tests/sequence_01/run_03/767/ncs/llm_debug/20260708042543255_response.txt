package ts01qwen3_7_plus;

import io.restassured.RestAssured;
import org.junit.Test;
import static org.hamcrest.Matchers.lessThan;

public class RemainderTest {

    private final String baseUrl = System.getenv("BASE_URL") != null ? System.getenv("BASE_URL") : "http://localhost:8080";

    @Test(timeout = 60000)
    public void testRemainderAZero() {
        RestAssured.given()
                .pathParam("a", 0)
                .pathParam("b", 5)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderBZero() {
        RestAssured.given()
                .pathParam("a", 5)
                .pathParam("b", 0)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveAPositiveB() {
        RestAssured.given()
                .pathParam("a", 10)
                .pathParam("b", 3)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderPositiveANegativeB() {
        RestAssured.given()
                .pathParam("a", 10)
                .pathParam("b", -3)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeAPositiveB() {
        RestAssured.given()
                .pathParam("a", -10)
                .pathParam("b", 3)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(200);
    }

    @Test(timeout = 60000)
    public void testRemainderNegativeANegativeB() {
        RestAssured.given()
                .pathParam("a", -10)
                .pathParam("b", -3)
                .when()
                .get(baseUrl + "/api/remainder/{a}/{b}")
                .then()
                .statusCode(200);
    }
}
