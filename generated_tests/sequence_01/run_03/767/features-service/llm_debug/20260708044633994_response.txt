package ts01qwen3_7_plus;

import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;

public class DuplicatedObjectExceptionTest {

    private String getBaseUrl() {
        String envUrl = System.getenv("BASE_URL");
        if (envUrl != null && !envUrl.isEmpty()) {
            return envUrl;
        }
        return System.getProperty("test.base.url", "http://localhost:8080");
    }

    @Test(timeout = 60000)
    public void testDuplicatedProductThrowsException() {
        String productName = "Product_" + UUID.randomUUID().toString();
        String baseUrl = getBaseUrl();

        given()
            .baseUri(baseUrl)
        .when()
            .post("/products/{productName}", productName)
        .then()
            .statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
        .when()
            .post("/products/{productName}", productName)
        .then()
            .statusCode(400);
    }

    @Test(timeout = 60000)
    public void testDuplicatedFeatureThrowsException() {
        String productName = "Product_" + UUID.randomUUID().toString();
        String featureName = "Feature_" + UUID.randomUUID().toString();
        String baseUrl = getBaseUrl();

        given()
            .baseUri(baseUrl)
        .when()
            .post("/products/{productName}", productName)
        .then()
            .statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
        .when()
            .post("/products/{productName}/features/{featureName}", productName, featureName)
        .then()
            .statusCode(lessThan(300));

        given()
            .baseUri(baseUrl)
        .when()
            .post("/products/{productName}/features/{featureName}", productName, featureName)
        .then()
            .statusCode(400);
    }
}
