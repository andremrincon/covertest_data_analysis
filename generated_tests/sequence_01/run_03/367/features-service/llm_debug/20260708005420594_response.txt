package ts01gpt_5_mini;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import io.restassured.http.ContentType;
import org.junit.BeforeClass;
import org.junit.Test;
import java.util.UUID;
import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class ProductTest {

    private static String BASE;

    @BeforeClass
    public static void setup() {
        String env = System.getenv("API_BASE_URL");
        if (env != null && !env.isEmpty()) {
            BASE = env;
        } else {
            BASE = System.getProperty("api.base", "http://localhost:8080");
        }
        RestAssured.useRelaxedHTTPSValidation();
    }

    @Test(timeout = 60000)
    public void testAddFeatureToProduct_shouldReturn201() {
        String product = "prod-" + UUID.randomUUID().toString();
        String feature = "feat-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "Test feature").when().post("/products/{productName}/features/{featureName}", product, feature);
        assertEquals(201, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testRemoveFeatureFromProduct_shouldReturn204() {
        String product = "prod-" + UUID.randomUUID().toString();
        String feature = "feat-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "to remove").when().post("/products/{productName}/features/{featureName}", product, feature).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).when().delete("/products/{productName}/features/{featureName}", product, feature);
        assertEquals(204, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetFeaturesForProduct_shouldReturn200() {
        String product = "prod-" + UUID.randomUUID().toString();
        String f1 = "featA-" + UUID.randomUUID().toString();
        String f2 = "featB-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "first").when().post("/products/{productName}/features/{featureName}", product, f1).then().statusCode(lessThan(300));
        given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "second").when().post("/products/{productName}/features/{featureName}", product, f2).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).when().get("/products/{productName}/features", product);
        assertEquals(200, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testUpdateFeatureOfProduct_shouldReturn200() {
        String product = "prod-" + UUID.randomUUID().toString();
        String feature = "featUpd-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "initial").when().post("/products/{productName}/features/{featureName}", product, feature).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).contentType(ContentType.URLENC).formParam("description", "updated description").when().put("/products/{productName}/features/{featureName}", product, feature);
        assertEquals(200, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddFeatureToConfiguration_shouldReturn201() {
        String product = "prod-" + UUID.randomUUID().toString();
        String config = "cfg-" + UUID.randomUUID().toString();
        String feature = "cfgFeat-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        given().baseUri(BASE).when().post("/products/{productName}/configurations/{configurationName}", product, config).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).when().post("/products/{productName}/configurations/{configurationName}/features/{featureName}", product, config, feature);
        assertEquals(201, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddRequiresConstraint_shouldReturn201() {
        String product = "prod-" + UUID.randomUUID().toString();
        String source = "src-" + UUID.randomUUID().toString();
        String required = "req-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).contentType(ContentType.URLENC).formParam("sourceFeature", source).formParam("requiredFeature", required).when().post("/products/{productName}/constraints/requires", product);
        assertEquals(201, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testAddExcludesConstraint_shouldReturn201() {
        String product = "prod-" + UUID.randomUUID().toString();
        String source = "srcEx-" + UUID.randomUUID().toString();
        String excluded = "excl-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).contentType(ContentType.URLENC).formParam("sourceFeature", source).formParam("excludedFeature", excluded).when().post("/products/{productName}/constraints/excludes", product);
        assertEquals(201, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testGetProductByName_shouldReturn200() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).when().get("/products/{productName}", product);
        assertEquals(200, r.getStatusCode());
    }

    @Test(timeout = 60000)
    public void testDeleteProduct_shouldReturn204() {
        String product = "prod-" + UUID.randomUUID().toString();
        given().baseUri(BASE).when().post("/products/{productName}", product).then().statusCode(lessThan(300));
        Response r = given().baseUri(BASE).when().delete("/products/{productName}", product);
        assertEquals(204, r.getStatusCode());
    }
}
