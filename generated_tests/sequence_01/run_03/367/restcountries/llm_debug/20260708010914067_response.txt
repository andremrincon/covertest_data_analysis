package ts01qwen3_7_plus;

import io.restassured.response.Response;
import org.junit.BeforeClass;
import org.junit.Test;

import static io.restassured.RestAssured.given;
import static org.hamcrest.Matchers.lessThan;
import static org.junit.Assert.assertEquals;

public class CORSFilterTest {

    private static String baseUrl;

    @BeforeClass
    public static void setup() {
        String envUrl = System.getenv("BASE_URL");
        baseUrl = (envUrl != null && !envUrl.isEmpty()) ? envUrl : System.getProperty("baseURI", "http://localhost:8080/rest");
    }

    @Test(timeout = 60000)
    public void testCorsAllowOriginOnV1All() {
        given().when().get(baseUrl + "/v1/all").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v1/all").then().extract().response();
        assertEquals("*", response.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowMethodsOnV1Alpha() {
        given().when().get(baseUrl + "/v1/alpha/US").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v1/alpha/US").then().extract().response();
        assertEquals("GET", response.getHeader("Access-Control-Allow-Methods"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowHeadersOnV1Currency() {
        given().when().get(baseUrl + "/v1/currency/USD").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v1/currency/USD").then().extract().response();
        assertEquals("Accept, X-Requested-With", response.getHeader("Access-Control-Allow-Headers"));
    }

    @Test(timeout = 60000)
    public void testCorsCacheControlOnV1Name() {
        given().when().get(baseUrl + "/v1/name/France").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v1/name/France").then().extract().response();
        assertEquals("public, max-age=86400", response.getHeader("Cache-Control"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowOriginOnV2All() {
        given().when().get(baseUrl + "/v2/all").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v2/all").then().extract().response();
        assertEquals("*", response.getHeader("Access-Control-Allow-Origin"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowMethodsOnV2Alpha() {
        given().when().get(baseUrl + "/v2/alpha/US").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v2/alpha/US").then().extract().response();
        assertEquals("GET", response.getHeader("Access-Control-Allow-Methods"));
    }

    @Test(timeout = 60000)
    public void testCorsAllowHeadersOnV2Currency() {
        given().when().get(baseUrl + "/v2/currency/EUR").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v2/currency/EUR").then().extract().response();
        assertEquals("Accept, X-Requested-With", response.getHeader("Access-Control-Allow-Headers"));
    }

    @Test(timeout = 60000)
    public void testCorsCacheControlOnV2Name() {
        given().when().get(baseUrl + "/v2/name/Germany").then().statusCode(lessThan(300));
        Response response = given().when().get(baseUrl + "/v2/name/Germany").then().extract().response();
        assertEquals("public, max-age=86400", response.getHeader("Cache-Control"));
    }
}
